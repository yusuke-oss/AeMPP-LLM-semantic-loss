"""
TPP-LLM Analysis and Visualization Tools
"""
import os
import numpy as np
import torch
import torch.nn.functional as F
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.metrics import confusion_matrix, classification_report, balanced_accuracy_score

def visualize_qualitative_sequence(raw_data, save_dir, phase_name, epoch, num_examples=5):
    """
    Generate trajectory plots for qualitative evaluation.
    Top: True vs Predicted Event Type (Discrete)
    Bottom: Absolute Cumulative Time (Lower bound = 0)
    """
    print(f"Generating qualitative sequence plots (Type & Abs Time) for epoch {epoch}...")
    
    sns.set_theme(style="whitegrid", context="talk", font_scale=1.1)
    selected_seqs = raw_data[:num_examples]
    
    type_to_y = {0: 2, 1: 1, 2: 0} 
    y_labels = {0: "Small", 1: "Medium", 2: "Large"}

    # Organize neatly into the "plots_sequence" directory
    vis_dir = os.path.join(save_dir, f"analysis/plots_sequence/{phase_name}")
    os.makedirs(vis_dir, exist_ok=True)

    for i, seq in enumerate(selected_seqs):
        true_types = np.array(seq['true_types']).flatten()
        pred_types = np.array(seq['pred_types']).flatten()
        true_times = np.array(seq['true_times']).flatten()
        pred_times = np.array(seq['pred_times']).flatten()

        min_len = min(len(true_types), len(pred_types), len(true_times), len(pred_times))
        true_types = true_types[:min_len]
        pred_types = pred_types[:min_len]
        true_times = true_times[:min_len]
        pred_times = pred_times[:min_len]

        try:
            true_types = true_types.astype(int)
            pred_types = pred_types.astype(int)
        except ValueError:
            pass 

        true_y = [type_to_y.get(t, t) for t in true_types]
        pred_y = [type_to_y.get(t, t) for t in pred_types]
        steps = np.arange(1, min_len + 1)

        acc_str = "N/A"
        if len(true_types) > 0:
            type_acc = np.mean(true_types == pred_types)
            acc_str = f"Type Acc:\n{type_acc:.1%}"

        rmse_str = "N/A"
        if len(true_times) > 0:
            time_rmse = np.sqrt(np.mean((true_times - pred_times) ** 2))
            rmse_str = f"Time RMSE:\n{time_rmse:.4f}"

        fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(12, 10), sharex=False)
        plt.subplots_adjust(right=0.85, hspace=0.15)

        # ==========================================
        # Upper plot: Event Types
        # ==========================================
        ax1.grid(True, axis='y', linestyle='--', alpha=0.7)
        ax1.grid(True, axis='x', linestyle=':', alpha=0.5)

        ax1.plot(steps, true_y, color='#1f77b4', marker='s', markersize=10, 
                 linestyle='-', linewidth=2, label='True Type', alpha=0.6)
        ax1.plot(steps, pred_y, color='#ff7f0e', marker='o', markersize=8, 
                 linestyle='--', linewidth=2, label='Predicted Type', alpha=0.9)

        yticks_vals = sorted(y_labels.keys())
        ax1.set_yticks(yticks_vals)
        ax1.set_yticklabels([y_labels[v] for v in yticks_vals])
        ax1.set_ylim(-0.5, 2.5)
        
        ax1.legend(loc='upper left', bbox_to_anchor=(1.02, 1.0), frameon=True, fontsize=12)

        props = dict(boxstyle='round', facecolor='white', alpha=1.0, edgecolor='gray')
        ax1.text(1.02, 0.0, acc_str, transform=ax1.transAxes, fontsize=12, 
                 verticalalignment='bottom', bbox=props)

        # ==========================================
        # Lower plot: Absolute Times
        # ==========================================
        ax2.grid(True, linestyle='--', alpha=0.7)
        ax2.plot(steps, true_times, color='black', marker='s', markersize=6,
                 linestyle='-', label='True Time', alpha=0.6)
        ax2.plot(steps, pred_times, color='green', marker='^', markersize=6,
                 linestyle='--', label='Predicted Time', alpha=0.8)
        ax2.set_ylim(bottom=0) 
        
        ax2.legend(loc='upper left', bbox_to_anchor=(1.02, 1.0), frameon=True, fontsize=12)

        ax2.text(1.02, 0.0, rmse_str, transform=ax2.transAxes, fontsize=12, 
                 verticalalignment='bottom', bbox=props)

        save_path = os.path.join(vis_dir, f"seq_{i}_type_time.png")
        plt.savefig(save_path, bbox_inches='tight')
        plt.close()

    print(f"Saved qualitative plots to {vis_dir}")

def perform_quantitative_analysis(
    all_types_true, all_types_pred,
    all_times_true, all_time_preds,
    all_time_deltas_true,
    result_save_path, phase_name,
    epoch, # Passed from the runner
    model=None  ,# 🌟 model を受け取れるように追加,
    seq_scores=None
):
    """
    Evaluate quantitative metrics (Confusion Matrix, RMSE, Classification Report).
    Results are saved neatly in the detailed analysis directory by epoch.
    """
    #print(f"--- Running Quantitative Analysis ({phase_name} - Epoch {epoch}) ---")
    
    # Create an epoch-specific directory within the detailed metrics folder
    quant_dir = os.path.join(result_save_path, "analysis", f"detail/epoch_{epoch}/{phase_name}")
    os.makedirs(quant_dir, exist_ok=True)

    # 1. Confusion Matrix
    try:
        # 【修正後】（0, 1, 2 の順に合わせる）
        class_names = ['Large', 'Medium', 'Small']
        cm = confusion_matrix(all_types_true, all_types_pred, labels=[0, 1, 2], normalize='true')
        sns.reset_orig()  # Reset seaborn settings to prevent layout issues
        import matplotlib as mpl
        mpl.rcParams.update({'font.size': 12, 'axes.titlesize': 14, 'axes.labelsize': 12})
        
        # --- Clean Version (Untitled for Academic Papers) ---
        fig, ax = plt.subplots(figsize=(6, 5))
        sns.heatmap(cm, annot=True, fmt=".2f", cmap="Blues", ax=ax,
                    xticklabels=class_names, yticklabels=class_names,
                    vmin=0.0, vmax=1.0, square=True, cbar_kws={"shrink": .8})
        ax.set_ylabel('True label', fontweight='bold')
        ax.set_xlabel('Predicted label', fontweight='bold')
        
        # Save with a simple filename
        plt.savefig(os.path.join(quant_dir, f'confusion_matrix.png'), 
                    bbox_inches='tight', dpi=150)
        plt.close(fig)
        
    except Exception as e:
        print(f"Failed to generate Confusion Matrix: {e}")

    # 2. Time Error Analysis
    try:
        bins_hours = [0, 1, 6, 12, np.inf]
        bin_labels = ['(0-1 hour)', '(1-6 hours)', '(6-12 hours)', '(12+ hours)']
        
        all_time_deltas_true_hours = all_time_deltas_true * 24.0
        bin_indices = np.digitize(all_time_deltas_true_hours, bins=bins_hours)
        
        analysis_log_time = f"--- {phase_name} Absolute Time RMSE by Intervals ---\n"
        
        for i in range(1, len(bins_hours)):
            label = bin_labels[i-1]
            mask = (bin_indices == i)
            
            if np.sum(mask) == 0:
                analysis_log_time += f"{label}: N=0\n"
                continue
                
            true_abs_times_in_bin = all_times_true[mask]
            pred_abs_times_in_bin = all_time_preds[mask]
            rmse_in_bin = np.sqrt(np.mean((true_abs_times_in_bin - pred_abs_times_in_bin) ** 2))
            
            analysis_log_time += f"{label}: N={np.sum(mask)}, RMSE={rmse_in_bin:.4f}\n"
    except Exception as e:
        analysis_log_time = f"Time Error Analysis Failed: {e}\n"

    # 3. Classification Report
    try:
        report = classification_report(all_types_true, all_types_pred, target_names=class_names, digits=4)
        bal_acc = balanced_accuracy_score(all_types_true, all_types_pred)
        analysis_log_report = f"--- {phase_name} Classification Report ---\n{report}\nBalanced Accuracy: {bal_acc:.4f}\n"
    except Exception as e:
        analysis_log_report = f"Classification Report Failed: {e}\n"
    
    # 4. Save Logs
    try:
        combined_log = analysis_log_time + "\n" + analysis_log_report
        #print(combined_log)
        
        with open(os.path.join(quant_dir, f'analysis_report.txt'), 'w') as f:
            f.write(combined_log)
            
    except Exception as e:
        print(f"Failed to save analysis logs: {e}")

    # 5. Call Sequence Qualitative Analysis
    if seq_scores and 'raw_data' in seq_scores:
        visualize_qualitative_sequence(seq_scores['raw_data'], result_save_path, phase_name, epoch, num_examples=30)
        # 🌟🌟🌟 追加: 新しく作ったCodebook単語可視化関数を呼び出す 🌟🌟🌟
        shared_sequence_indices =visualize_codebook_selection(seq_scores['raw_data'], result_save_path, phase_name, epoch, num_examples=10, max_events=13)
        # --- perform_quantitative_analysis の一番最後に追加 ---
        # 🌟🌟🌟 ここに追加：単語の多様性（全体集計）を実行 🌟🌟🌟
        analyze_vocabulary_diversity(seq_scores['raw_data'], result_save_path, phase_name, epoch)
        # 🌟🌟🌟 追加: t-SNEによる軌跡可視化を実行 🌟🌟🌟
        visualize_codebook_tsne_trajectory(
            seq_scores['raw_data'], result_save_path, phase_name, epoch, model=model, num_examples=len(shared_sequence_indices),max_events=14,selected_indices=shared_sequence_indices
        )
        validate_codebook_tsne_alignment(
            seq_scores['raw_data'],
            result_save_path,
            phase_name,
            epoch,
            selected_indices=shared_sequence_indices,
            codebook_max_events=14,
            tsne_max_events=14
        )
        # 🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟🌟
        
        import json
        # 🌟 "test" と "initial_test" (load_flagのみの実行時) の両方に反応させる
        if "test" in phase_name:
            save_data = {
                "lls": seq_scores.get("lls", []),
                "rmses": seq_scores.get("rmses", []),
                "f1s": seq_scores.get("f1s", [])
            }
            out_path = os.path.join(result_save_path, "test_seq_scores.json")
            with open(out_path, "w") as f:
                json.dump(save_data, f)
            print(f"✅ Saved sequence scores for paired t-test to {out_path}")

def analyze_vocabulary_diversity(raw_data, save_dir, phase_name, epoch):
    """
    全シーケンス・全イベントを通じて、どの単語が何回アンカーとして選ばれたかを
    集計し、頻度分布（ヒストグラム）として保存する。
    """
    print(f"📊 Analyzing vocabulary diversity for {phase_name} (Epoch {epoch})...")
    
    from collections import Counter
    import matplotlib.pyplot as plt
    import os

    # 保存先ディレクトリの作成
    div_dir = os.path.join(save_dir, f"analysis/vocabulary_diversity/{phase_name}")
    os.makedirs(div_dir, exist_ok=True)

    all_words_p = []
    all_words_h = []

    # 全シーケンスの全イベントから、選ばれた単語をフラットなリストに集約
    for seq in raw_data:
        if 'words_p' in seq and seq['words_p']:
            all_words_p.extend(seq['words_p'])
        if 'words_h' in seq and seq['words_h']:
            all_words_h.extend(seq['words_h'])

    # 集計とプロットを行う内部関数
    def plot_and_save_counter(words_list, label_name):
        if not words_list:
            return

        # N/A などの不要な文字を除去し、綺麗にカウント
        cleaned_words = [str(w).replace('Ġ', '').replace(' ', '') for w in words_list if w != "N/A"]
        word_counts = Counter(cleaned_words)
        
        total_events = len(cleaned_words)
        unique_words = len(word_counts)
        
        # 出現回数順にソート
        sorted_counts = word_counts.most_common()

        # --- テキストログとして保存 ---
        log_path = os.path.join(div_dir, f"epoch_{epoch}_{label_name}_diversity_log.txt")
        with open(log_path, "w", encoding="utf-8") as f:
            f.write(f"=== Vocabulary Diversity Report ({label_name}) ===\n")
            f.write(f"Total Evaluated Events : {total_events}\n")
            f.write(f"Unique Words Selected  : {unique_words} words\n")
            f.write(f"--------------------------------------------------\n")
            f.write(f"{'Rank':<5} | {'Word':<20} | {'Count':<8} | {'Ratio (%)':<8}\n")
            f.write(f"--------------------------------------------------\n")
            for i, (word, count) in enumerate(sorted_counts):
                ratio = (count / total_events) * 100
                f.write(f"{i+1:<5} | {word:<20} | {count:<8} | {ratio:.2f}%\n")

        # --- ヒストグラム（棒グラフ）として保存 ---
        # 上位30単語だけをプロット
        top_k = min(30, unique_words)
        plot_words = [item[0] for item in sorted_counts[:top_k]]
        plot_counts = [item[1] for item in sorted_counts[:top_k]]

        fig, ax = plt.subplots(figsize=(12, 6))
        ax.bar(plot_words, plot_counts, color='skyblue', edgecolor='black')
        
        ax.set_title(f"Top {top_k} Selected Words in {label_name} (Epoch {epoch})\nTotal Events: {total_events} | Unique Words: {unique_words}", fontsize=14)
        ax.set_xlabel("Selected Word", fontsize=12)
        ax.set_ylabel("Frequency (Hits)", fontsize=12)
        
        # x軸のラベルを回転させて見やすくする
        plt.xticks(rotation=45, ha='right')
        ax.grid(axis='y', linestyle='--', alpha=0.7)
        plt.tight_layout()

        img_path = os.path.join(div_dir, f"epoch_{epoch}_{label_name}_diversity_plot.png")
        plt.savefig(img_path, dpi=150)
        plt.close(fig)

    # Input(P) と Hidden(H) それぞれで実行
    plot_and_save_counter(all_words_p, "Input_P_Layer")
    plot_and_save_counter(all_words_h, "Hidden_H_Layer")
    
    print(f"✅ Vocabulary diversity analysis saved to {div_dir}")

def visualize_codebook_selection(raw_data, save_dir, phase_name, epoch, num_examples=5, max_events=40):

    else:

def visualize_codebook_tsne_trajectory(
    raw_data,
    save_dir,
    phase_name,
    epoch,
    model=None,
    codebook_embs=None,
    codebook_words=None,
    num_examples=5,
    max_events=30,
    use_hidden_codebook=False,
    perplexity=15,
    annot_top_words=100,
    selected_indices=None,
    local_pca_k=15,
    mds_n_init=4
):
    """
    Saved raw_data['input_vector'] / raw_data['hidden_vector'] を直接使用して
    t-SNE による Event 軌跡と Codebook トークンの可視化を行う。
    """

    import os
    import numpy as np
    import torch
    import matplotlib.pyplot as plt
    import traceback

    from sklearn.manifold import TSNE

    # ============================================================
    # 保存先
    # ============================================================
    vis_dir = os.path.join(
        save_dir,
        f"analysis/plots_tsne/{phase_name}/epoch_{epoch}"
    )

    os.makedirs(
        vis_dir,
        exist_ok=True
    )

    print(
        "\n============================================================"
    )
    print(
        "🎨 [Embedding Visualization (t-SNE Only)]"
    )
    print(
        f"Phase : {phase_name}"
    )
    print(
        f"Epoch : {epoch}"
    )
    print(
        "Mode  : t-SNE"
    )
    print(
        "============================================================"
    )

    # ============================================================
    # DDP 対応
    # ============================================================
    base_model = (
        model.module
        if (
            model is not None
            and hasattr(model, "module")
        )
        else model
    )

    # ============================================================
    # Codebook P / H
    # ============================================================
    codebook_p = None
    codebook_h = None

    codebook_words_p = None
    codebook_words_h = None

    # ============================================================
    # model から codebook を取得
    # ============================================================
    if base_model is not None:

        if (
            hasattr(base_model, "codebook_p")
            and base_model.codebook_p is not None
        ):
            codebook_p = (
                base_model
                .codebook_p
                .detach()
                .cpu()
                .numpy()
                .astype(np.float32)
            )

        if (
            hasattr(base_model, "codebook_h")
            and base_model.codebook_h is not None
        ):
            codebook_h = (
                base_model
                .codebook_h
                .detach()
                .cpu()
                .numpy()
                .astype(np.float32)
            )

        codebook_words_p = getattr(
            base_model,
            "codebook_words_p",
            None
        )
        codebook_words_h = getattr(
            base_model,
            "codebook_words_h",
            None
        )

        generic_words = getattr(
            base_model,
            "codebook_words",
            []
        )

        if codebook_words_p is None:
            codebook_words_p = generic_words

        if codebook_words_h is None:
            codebook_words_h = generic_words

        if codebook_p is None:
            if (
                hasattr(base_model, "codebook")
                and base_model.codebook is not None
            ):
                codebook_p = (
                    base_model
                    .codebook
                    .detach()
                    .cpu()
                    .numpy()
                    .astype(np.float32)
                )

        if codebook_h is None:
            if (
                hasattr(base_model, "codebook")
                and base_model.codebook is not None
            ):
                codebook_h = (
                    base_model
                    .codebook
                    .detach()
                    .cpu()
                    .numpy()
                    .astype(np.float32)
                )

    # ============================================================
    # 明示的 codebook_embs
    # ============================================================
    if codebook_embs is not None:
        if isinstance(
            codebook_embs,
            torch.Tensor
        ):
            codebook_embs = (
                codebook_embs
                .detach()
                .cpu()
                .numpy()
            )

        codebook_embs = np.asarray(
            codebook_embs,
            dtype=np.float32
        )

        if use_hidden_codebook:
            codebook_h = codebook_embs
        else:
            codebook_p = codebook_embs

    # ============================================================
    # 明示的 codebook_words
    # ============================================================
    if codebook_words is not None:
        if use_hidden_codebook:
            codebook_words_h = codebook_words
        else:
            codebook_words_p = codebook_words

    # ============================================================
    # selected_indices
    # ============================================================
    if selected_indices is None:
        print(
            "⚠️ [Embedding Visualization] "
            "selected_indices が指定されていません。"
        )
        return []

    print(
        f"🔗 [Embedding Visualization] "
        f"selected_indices = {selected_indices}"
    )

    # ============================================================
    # mode
    # ============================================================
    modes = [
        {
            "layer_name": "Input (P)",
            "suffix": "input",
            "vector_key": "input_vector",
            "codebook": codebook_p,
            "codebook_words": codebook_words_p,
        },
        {
            "layer_name": "Hidden (H)",
            "suffix": "hidden",
            "vector_key": "hidden_vector",
            "codebook": codebook_h,
            "codebook_words": codebook_words_h,
        }
    ]

    plotted_results = []

    # ============================================================
    # Sequence Loop
    # ============================================================
    for orig_idx in selected_indices:

        if (
            orig_idx < 0
            or orig_idx >= len(raw_data)
        ):
            print(f"⚠️ Invalid sequence index: {orig_idx}")
            continue

        seq = raw_data[orig_idx]

        raw_times = list(seq.get("time", seq.get("tsne_time", [])))
        raw_types = list(seq.get("type", seq.get("tsne_type", [])))

        words_p = seq.get("words_p", [])
        words_h = seq.get("words_h", [])

        # ========================================================
        # Input / Hidden
        # ========================================================
        for mode in modes:

            layer_name = mode["layer_name"]
            suffix = mode["suffix"]
            vector_key = mode["vector_key"]
            cb_embs = mode["codebook"]
            cb_words = mode["codebook_words"]

            tsne_png = None
            tsne_pdf = None

            if cb_embs is None:
                print(f"⚠️ Seq {orig_idx} | {layer_name}: Codebook がありません。")
                continue

            cb_embs = np.asarray(cb_embs, dtype=np.float32)

            if cb_embs.ndim != 2:
                print(f"⚠️ Seq {orig_idx} | {layer_name}: Codebook shape={cb_embs.shape}")
                continue

            # ====================================================
            # ★ Input (P) のときだけ、重複する単語・ベクトルをユニークにする
            # ====================================================
            if suffix == "input" and cb_words is not None:
                seen_words = set()
                unique_indices = []
                for idx, w in enumerate(cb_words):
                    w_clean = str(w).replace("Ġ", "").replace(" ", "")
                    if w_clean not in seen_words:
                        seen_words.add(w_clean)
                        unique_indices.append(idx)
                
                cb_embs = cb_embs[unique_indices]
                cb_words = [cb_words[i] for i in unique_indices]

            saved_vectors = seq.get(vector_key, None)

            if saved_vectors is None:
                print(f"⚠️ Seq {orig_idx} | {layer_name}: raw_data['{vector_key}'] がありません。")
                continue

            try:
                event_embs = np.asarray(saved_vectors, dtype=np.float32)
            except Exception as e:
                print(f"❌ Seq {orig_idx} | {layer_name}: {vector_key} conversion error: {e}")
                continue

            if event_embs.ndim == 1:
                event_embs = event_embs.reshape(1, -1)

            if event_embs.ndim != 2:
                print(f"⚠️ Seq {orig_idx} | {layer_name}: vector shape={event_embs.shape}")
                continue

            if event_embs.shape[1] != cb_embs.shape[1]:
                print(
                    f"⚠️ Seq {orig_idx} | {layer_name}: "
                    f"dimension mismatch: event={event_embs.shape[1]}, codebook={cb_embs.shape[1]}"
                )
                continue

            event_lengths = [len(event_embs), len(raw_times), len(raw_types)]
            if suffix == "input":
                if len(words_p) > 0:
                    event_lengths.append(len(words_p))
            else:
                if len(words_h) > 0:
                    event_lengths.append(len(words_h))

            raw_event_count = min(event_lengths)
            n_events = min(raw_event_count, max_events)

            if n_events <= 0:
                print(f"⚠️ Seq {orig_idx} | {layer_name}: Event がありません。")
                continue

            ev_embs = event_embs[:n_events]

            # ====================================================
            # t-SNE 処理
            # ====================================================
            combined_embs = np.vstack([cb_embs, ev_embs])
            num_codebook_tokens = cb_embs.shape[0]
            total_samples = combined_embs.shape[0]

            if total_samples > 2:
                effective_perplexity = max(
                    1,
                    min(perplexity, total_samples - 1)
                )

                try:
                    tsne = TSNE(
                        n_components=2,
                        perplexity=effective_perplexity,
                        random_state=42,
                        init="pca",
                        learning_rate="auto"
                    )

                    embedded_2d = tsne.fit_transform(combined_embs)

                    cb_2d = embedded_2d[:num_codebook_tokens]
                    ev_2d = embedded_2d[num_codebook_tokens:]

                    fig, ax = plt.subplots(figsize=(14, 10))
                    ax.grid(True, linestyle="--", alpha=0.4)

                    # Codebook scatter
                    ax.scatter(
                        cb_2d[:, 0],
                        cb_2d[:, 1],
                        c="#a0c4ff",
                        alpha=0.45,
                        s=60,
                        edgecolors="#4361ee",
                        label="Codebook Tokens",
                        zorder=2
                    )

                    # ====================================================
                    # Codebook words annotations
                    # Input (P): 従来どおり単語名を表示
                    # Hidden (H): 重複単語に番号を付けて区別
                    # ====================================================
                    if (
                        cb_words is not None
                        and len(cb_words) == num_codebook_tokens
                    ):
                        display_words = []
                        word_counts = {}

                        for w in cb_words:
                            w_clean = str(w).replace("Ġ", "").replace(" ", "")

                            if suffix == "hidden":
                                word_counts[w_clean] = word_counts.get(w_clean, 0) + 1
                                count = word_counts[w_clean]

                                if count == 1:
                                    display_word = w_clean
                                else:
                                    display_word = f"{w_clean} ({count})"
                            else:
                                display_word = w_clean

                            display_words.append(display_word)

                        for idx, word_label in enumerate(display_words):
                            x_pos = cb_2d[idx, 0]
                            y_pos = cb_2d[idx, 1]

                            ax.annotate(
                                word_label,
                                xy=(x_pos, y_pos),
                                xytext=(x_pos + 0.3, y_pos + 0.3),
                                fontsize=8,
                                color="#1d3557",
                                weight="bold",
                                bbox=dict(
                                    boxstyle="round,pad=0.2",
                                    fc="white",
                                    ec="#a0c4ff",
                                    alpha=0.85,
                                    lw=0.6
                                ),
                                arrowprops=dict(
                                    arrowstyle="-",
                                    color="#a0c4ff",
                                    lw=0.8,
                                    shrinkA=0,
                                    shrinkB=4
                                ),
                                zorder=3
                            )


                    try:
                        cmap = plt.colormaps["plasma"]
                    except AttributeError:
                        cmap = plt.cm.get_cmap("plasma")

                    # Event trajectory line
                    if n_events >= 2:
                        ax.plot(
                            ev_2d[:, 0],
                            ev_2d[:, 1],
                            color="#333333",
                            linestyle="--",
                            linewidth=1.8,
                            alpha=0.7,
                            zorder=4,
                            label="Event Trajectory"
                        )

                    # Direction arrows
                    for j in range(n_events - 1):
                        ax.annotate(
                            "",
                            xy=(ev_2d[j + 1, 0], ev_2d[j + 1, 1]),
                            xytext=(ev_2d[j, 0], ev_2d[j, 1]),
                            arrowprops=dict(
                                arrowstyle="->",
                                color="black",
                                lw=1.5,
                                alpha=0.8,
                                mutation_scale=12
                            ),
                            zorder=4
                        )

                    # Event points / IDs
                    for j in range(n_events):
                        event_id = f"E{j:d}"

                        ax.scatter(
                            ev_2d[j, 0], ev_2d[j, 1],
                            facecolors="none", s=180, marker="X",
                            edgecolors="black", lw=0.8, zorder=5
                        )
                            

                        ax.text(
                            ev_2d[j, 0] + 0.3,
                            ev_2d[j, 1] + 0.3,
                            event_id,
                            fontsize=8,
                            fontweight="bold",
                            color="#2b2d42",
                            bbox=dict(
                                boxstyle="round,pad=0.2",
                                fc="#fff3bf",
                                ec="#f4a261",
                                alpha=0.9,
                                lw=0.8
                            ),
                            zorder=7
                        )

                    #ax.set_title(
                    #    f"Event Embedding Trajectory (t-SNE)\n"
                    #    f"Seq {orig_idx}, Epoch {epoch} | Phase: {phase_name} | Layer: {layer_name} | Events: {n_events}",
                    #    fontsize=13,
                    #    fontweight="bold"
                    #)
                    ax.set_xlabel("t-SNE Dimension 1", fontsize=12)
                    ax.set_ylabel("t-SNE Dimension 2", fontsize=12)
                    ax.legend(loc="upper right", frameon=True, facecolor="white", framealpha=0.9, fontsize=10)

                    # --- 枠線・グリッドを消す設定 ---
                    #plt.grid(False)  # 方眼紙状のグリッド線を消す

                    # もし背景色（グレー）自体を真っ白にしたい場合
                    #ax = plt.gca()
                    #ax.set_facecolor("white")  # 背景色を白に設定
                    # 軸（目盛りと数字）を丸ごと非表示にする
                    #plt.axis("off")

                    tsne_png = os.path.join(vis_dir, f"tsne_trajectory_seq{orig_idx}_{suffix}.png")
                    tsne_pdf = os.path.join(vis_dir, f"tsne_trajectory_seq{orig_idx}_{suffix}.pdf")

                    plt.savefig(tsne_png, bbox_inches="tight", dpi=150)
                    plt.savefig(tsne_pdf, bbox_inches="tight")
                    plt.close(fig)

                    print(f"✅ [t-SNE] Seq {orig_idx} | {layer_name} | Events={n_events}")

                except Exception as e:
                    print(f"❌ [t-SNE] Seq {orig_idx} | {layer_name}: {e}")
                    traceback.print_exc()
            else:
                print(f"⚠️ [t-SNE] Seq {orig_idx} | {layer_name}: sample数不足のためスキップ。")

            # Result 記録 (チェックツールが求めるキーに対応)
            res_dict = {
                "seq_idx": orig_idx,
                "mode": suffix,
                "event_count": n_events,
                "vector_key": vector_key,
                "tsne_png": tsne_png,
                "tsne_pdf": tsne_pdf,
            }
            if suffix == "input":
                res_dict["input_tsne"] = tsne_png if (tsne_png and os.path.exists(tsne_png)) else False
            else:
                res_dict["hidden_tsne"] = tsne_png if (tsne_png and os.path.exists(tsne_png)) else False

            plotted_results.append(res_dict)

    # ============================================================
    # 終了
    # ============================================================
    print(
        "\n============================================================"
    )
    print(
        f"✅ [Embedding Visualization] Saved {len(plotted_results)} results (t-SNE only)"
    )
    print(
        "============================================================"
    )

    return plotted_results

def validate_codebook_tsne_alignment(
    raw_data,
    save_dir,
    phase_name,
    epoch,
    selected_indices,
    codebook_max_events=13,
    tsne_max_events=13
):
    """
    Codebook / t-SNE / raw event の対応関係を検証する。

    重要:
        true_times / true_types は prediction 用に [1:] されているため、
        元イベント数の判定には使わない。

    元イベント列:
        time
        type
        tsne_time
        tsne_type
        input_vector
        hidden_vector
        words_p
        words_h

    評価列:
        true_times
        true_types
        pred_times
        pred_types

    検証内容:
        1. raw event 各列の長さ一致
        2. words_p / words_h の長さ一致
        3. raw time == tsne_time
        4. raw type == tsne_type
        5. true_times == time[1:]
        6. true_types == type[1:]
        7. prediction/evaluation 列の長さが raw_event_count - 1
        8. Codebook / t-SNE の表示イベント数一致
        9. Codebook / Input t-SNE / Hidden t-SNE ファイル存在
    """

    import os
    import glob
    import csv
    import numpy as np

    # ============================================================
    # 保存先
    # ============================================================
    check_dir = os.path.join(
        save_dir,
        f"analysis/plots_tsne/{phase_name}/epoch_{epoch}",
        "alignment_check"
    )

    os.makedirs(
        check_dir,
        exist_ok=True
    )

    codebook_dir = os.path.join(
        save_dir,
        f"analysis/plots_codebook/{phase_name}/epoch_{epoch}"
    )

    tsne_dir = os.path.join(
        save_dir,
        f"analysis/plots_tsne/{phase_name}/epoch_{epoch}"
    )

    report_rows = []

    # ============================================================
    # Helper
    # ============================================================
    def _safe_len(x):

        if x is None:
            return 0

        try:
            return len(x)
        except Exception:
            return 0

    def _to_numpy(x):

        if x is None:
            return None

        try:
            return np.asarray(
                x,
                dtype=np.float32
            )
        except Exception:
            return None

    # ============================================================
    # 各系列
    # ============================================================
    for orig_idx in selected_indices:

        # --------------------------------------------------------
        # Index check
        # --------------------------------------------------------
        if (
            orig_idx < 0
            or orig_idx >= len(raw_data)
        ):

def float32_to_bytes_big_endian(value):
    bytes_obj = np.array([value], dtype=np.float32).tobytes()
    return bytes_obj[3], bytes_obj[2], bytes_obj[1], bytes_obj[0]

def float32_to_byte_tokens(value):
    int1, int2, int3, int4 = float32_to_bytes_big_endian(value)
    return f"<|byte_{int1}|><|byte_{int2}|><|byte_{int3}|><|byte_{int4}|>"

def float32_list_to_byte_token_list(values):
    return [float32_to_byte_tokens(v) for v in values]

def get_collate_fn(stats=None):
    """
    stats（統計量）を受け取って、標準化機能付きの collate_fn を返すファクトリ関数
    """
    def collate_fn(batch: dict) -> dict:
        # 返却用の辞書を準備
        collated = {
            'time_since_start': [torch.FloatTensor(item['time_since_start']) for item in batch],
            'time_since_last_event': [torch.FloatTensor(item['time_since_last_event']) for item in batch],
            'type_event': [torch.LongTensor(item['type_event']) for item in batch],
            'type_text': [item['type_text'] for item in batch],
            'magnitude' : [torch.FloatTensor(item['magnitude']) for item in batch],
            'depth': [torch.FloatTensor(item['depth']) for item in batch],
            'latitude' : [torch.FloatTensor(item['latitude']) for item in batch],
            'longitude': [torch.FloatTensor(item['longitude']) for item in batch],
        }

        # stats が渡されている場合は、標準化を行ってからバイトトークン化する
        if stats is not None:
            time_mean, time_std = stats['time']['mean'], stats['time']['std']
            mag_mean, mag_std = stats['mag']['mean'], stats['mag']['std']
            dep_mean, dep_std = stats['dep']['mean'], stats['dep']['std']
            lat_mean, lat_std = stats['lat']['mean'], stats['lat']['std']
            lon_mean, lon_std = stats['lon']['mean'], stats['lon']['std']
            
            time_tokens = []
            mag_tokens = []
            dep_tokens = []
            lat_tokens = []
            lon_tokens = []

            for item in batch:
                # 1. 生の数値を取得
                raw_time = np.array(item['time_since_start'], dtype=np.float32)
                raw_mag = np.array(item['magnitude'], dtype=np.float32)
                raw_dep = np.array(item['depth'], dtype=np.float32)
                raw_lat = np.array(item['latitude'], dtype=np.float32)
                raw_lon = np.array(item['longitude'], dtype=np.float32)

                # 2. 標準化（ゼロ除算防止に 1e-8 を足す）
                norm_time = (raw_time - time_mean) / (time_std + 1e-8)
                norm_mag = (raw_mag - mag_mean) / (mag_std + 1e-8)
                norm_dep = (raw_dep - dep_mean) / (dep_std + 1e-8)
                norm_lat = (raw_lat - lat_mean) / (lat_std + 1e-8)
                norm_lon = (raw_lon - lon_mean) / (lon_std + 1e-8)

                # 3. 標準化された数値をバイトトークンリストに変換！
                time_tokens.append(float32_list_to_byte_token_list(norm_time))
                mag_tokens.append(float32_list_to_byte_token_list(norm_mag))
                dep_tokens.append(float32_list_to_byte_token_list(norm_dep))
                lat_tokens.append(float32_list_to_byte_token_list(norm_lat))
                lon_tokens.append(float32_list_to_byte_token_list(norm_lon))

            collated['time_float'] = time_tokens       # 標準化済みのバイトトークン文字列
            collated['magnitude_float'] = mag_tokens  # 標準化済みのバイトトークン文字列
            collated['depth_float'] = dep_tokens      # 標準化済みのバイトトークン文字列
            collated['latitude_float'] = lat_tokens  # 標準化済みのバイトトークン文字列
            collated['longitude_float'] = lon_tokens      # 標準化済みのバイトトークン文字列
            
        return collated

    return collate_fn

def create_few_shot_dataset(data_dir, output_dir, few_shot_ratio=0.1, seed=0) -> None:
    random.seed(seed)
    train_file = os.path.join(data_dir, 'train.json')
    dev_file = os.path.join(data_dir, 'dev.json')
    test_file = os.path.join(data_dir, 'test.json')
    with open(train_file, 'r') as f:
        train_data = json.load(f)
    few_shot_size = int(len(train_data) * few_shot_ratio)
    few_shot_train_data = random.sample(train_data, few_shot_size)
    os.makedirs(output_dir, exist_ok=True)
    with open(os.path.join(output_dir, 'train.json'), 'w') as file:
        json.dump(few_shot_train_data, file, indent=4)
    shutil.copy(dev_file, os.path.join(output_dir, 'dev.json'))
    shutil.copy(test_file, os.path.join(output_dir, 'test.json'))
    print(f"Few-shot dataset created in {output_dir}.")

import math

class TimePositionalEncoding(nn.Module):
    """
    Temporal Positional Encoding from THP
    """

    def __init__(self, embedding_dim: int, dtype=torch.float32, device: Union[str, torch.device] = 'cpu'):
        """
        Initialize the temporal encoding

        :param embedding_dim: embedding dimension
        :param dtype: data type for the output
        :param device: device
        """
        super().__init__()
        self.dtype = dtype
        self.device = device
        i = torch.arange(0, embedding_dim, 1, dtype=self.dtype, device=self.device)
        div_term = (2 * (i // 2).to(self.dtype) * -(math.log(10000.0) / embedding_dim)).exp()
        self.register_buffer('div_term', div_term)

    def forward(self, event_times: Tensor) -> Tensor:
        """
        Compute time positional encoding defined in the THP model

        :param event_times: event times, (seq_len, 1)
        :return: temporal encoding vector, (seq_len, embedding_dim)
        """
        result = (event_times * self.div_term).to(self.dtype).to(self.device)
        result[:, 0::2] = torch.sin(result[:, 0::2])
        result[:, 1::2] = torch.cos(result[:, 1::2])
        return result

class MagnitudeLDLLoss(nn.Module):
    def __init__(
        self,
        softness=1.0,
        class_weights=None,
        reduction="sum"
    ):
        super().__init__()

        self.softness = softness

        self.criterion = nn.CrossEntropyLoss(
            weight=class_weights,
            reduction=reduction
        )

    def forward(self, logits, labels):
        labels = labels.to(
            device=logits.device,
            dtype=torch.long
        )

        # クラス順: Large=0, Medium=1, Small=2
        num_classes = logits.size(-1)

        class_ids = torch.arange(
            num_classes,
            device=logits.device,
            dtype=logits.dtype
        )

        # 正解クラスID k* を中心とする正規分布
        distances = (
            class_ids - labels.unsqueeze(-1).to(logits.dtype)
        )

        scores = -distances.pow(2) / (
            2.0 * self.softness**2
        )

        # Soft target label を生成
        target_dist = torch.softmax(scores, dim=-1)

        return self.criterion(
            logits,
            target_dist
        )

    def _init_concept_anchors(self):
        """
        Initialize and freeze the semantic anchors based on the word embeddings of specific concepts.
        """
        print("Initializing Concept Anchors...")
        
        txt_mag = "Magnitude"  
        txt_dep = "Depth"      
        txt_time = "Time"      
        
        with torch.no_grad():
            vec_mag = self.embed_event_text([txt_mag])[0].mean(dim=0)
            vec_dep = self.embed_event_text([txt_dep])[0].mean(dim=0)
            vec_time = self.embed_event_text([txt_time])[0].mean(dim=0)

        # Register as fixed buffers (no gradients will be computed)
        self.register_buffer('anchor_mag', vec_mag)
        self.register_buffer('anchor_dep', vec_dep)
        self.register_buffer('anchor_time', vec_time)
        print("Anchors Initialized (Fixed to Word Embeddings).")

    def embed_event_text(self, event_texts: List[str], add_special_tokens: bool = False) -> List[Tensor]:
        """
        Embed the texts of events

        :param event_texts: event texts
        :param add_special_tokens: add special tokens or not
        :return: token embeddings of event texts, [(text_token_len, embedding_dim), ...]
        """
        event_tokens = self.tokenizer(
            event_texts, return_tensors='pt', add_special_tokens=add_special_tokens,
            padding=True, truncation=False)
        
        nums_tokens = event_tokens['attention_mask'].to(self.device).sum(dim=-1)
        event_embeddings_padded = self.llm_embedder(event_tokens['input_ids'].to(self.device))
        event_embeddings = [
            event_embedding_padded[:num_tokens]
            for num_tokens, event_embedding_padded in zip(nums_tokens, event_embeddings_padded)]
        
        return event_embeddings
    
    def forward(
        self, batch_event_times: List[Tensor], 
        batch_event_texts: List[List[str]],batch_event_mag:List[Tensor],batch_event_dep:List[Tensor],batch_event_lat:List[Tensor],batch_event_lon:List[Tensor],batch_event_types: List[Tensor] = None) -> Tuple[List[Tensor]]:
        """
        Forward function to get hidden states of event sequences in a batch

        :param batch_event_times: batch of event times in sequences, [(seq_len,), ...]
        :param batch_event_time_deltas: batch of event time deltas in sequences, [(seq_len,), ...]
        :param batch_event_texts: batch of event texts in sequences, [(seq_len,), ...]
        :return: hidden states of events, [(seq_len, hidden_size), ...]

        batch_attention_masks = []
        batch_event_emb_indices = []
        batch_event_emb_indices_hidden = []

        
        if self.temporal_emb_type in [
            'MLP_token_renew_loss', 'MLP_token_renew_loss_p','MLP_token_renew_loss_ph','MLP_token_renew_loss_ph_label',
            'MLP_token_renew_topk_loss_p', 'MLP_token_renew_topk_loss_ph','MLP_token_renew_vocab_loss_ph' # ⬅️ 追加
        ]:
            self.temp_embs_for_loss = []

        # Process each event sequence in the batch
        for event_times, event_texts, event_mag, event_dep,  event_lat, event_lon, event_types in zip(
            batch_event_times, batch_event_texts, batch_event_mag, batch_event_dep,batch_event_lat, batch_event_lon, batch_event_types
        ):
            # =============== 追加: 数値データの正規化 (Min-Max Scaling) ===============
            # ゼロ除算を回避するために 1e-8 を加算して 0.0〜1.0 の範囲に正規化します
            #norm_event_times = (event_times - self.min_time) / (self.max_time - self.min_time + 1e-8)
            #norm_event_mag = (event_mag - self.min_mag) / (self.max_mag - self.min_mag + 1e-8)
            #norm_event_dep = (event_dep - self.min_dep) / (self.max_dep - self.min_dep + 1e-8)
            # =========================================================================

            # model.py の 324行目付近
            # =============== 変更: 数値データの標準化 (Standardization) ===============
            # ゼロ除算を回避するために 1e-8 を加算
            if self.temporal_emb_type in ['float']:
                pass
            else:
                norm_event_times = (event_times - self.mean_time) / (self.std_time + 1e-8)
                norm_event_mag = (event_mag - self.mean_mag) / (self.std_mag + 1e-8)
                norm_event_dep = (event_dep - self.mean_dep) / (self.std_dep + 1e-8)
                norm_event_lat = (event_lat - self.mean_lat) / (self.std_lat + 1e-8)
                norm_event_lon = (event_lon - self.mean_lon) / (self.std_lon + 1e-8)
            # =========================================================================

            prompt_token_embeddings_tensor = self.embed_event_text(event_texts=[self.prompt], add_special_tokens=True)[0]
            
            sequence_embeddings = []
            sequence_attention_mask = []
            event_emb_indices = []
            event_emb_indices_hidden = []

            if self.use_tokens:
                if self.use_natural_separators:
                    time_event_texts=["Time"]
                    type_event_texts=["Type"]
                    mag_event_texts=["Magnitude"]
                    dep_event_texts=["Depth"]
                    lat_event_texts=["Latitude"]
                    lon_event_texts=["Longitude"]
                    fusion_event_texts=["State"]
                    
                    
                else:

                    type_event_texts=["<|type_prefix|>"]
                    mag_event_texts=["<|magnitude_prefix|>"]
                    dep_event_texts=["<|depth_prefix|>"]
                    lat_event_texts=["<|latitude_prefix|>"]
                    lon_event_texts=["<|longitude_prefix|>"]
            
            
            
            # Add the prompt embeddings
            for prompt_token_embedding in prompt_token_embeddings_tensor :

                sequence_attention_mask.append(1)

            # =============== 変更: 埋め込み層への入力 (正規化された値を使用) ===============

                # ※ self.anchor_time などは _init_concept_anchors で取得済みの前提

                time_cat = torch.cat([exp_anchor_time, time_scalar], dim=-1)

                temporal_embeddings  = self.shared_mlp(time_cat)
                event_mag_embeddings = self.shared_mlp(mag_cat)

                seq_len = time_scalar.size(0)

                time_cat = torch.cat([exp_time_onehot, time_scalar], dim=-1)

                dep_cat  = torch.cat([exp_dep_onehot, dep_scalar], dim=-1)

                # ④ 共有MLP（Shared MLP）にすべて通して、それぞれの埋め込みベクトルを取得
                temporal_embeddings  = self.shared_mlp(time_cat)
                event_mag_embeddings = self.shared_mlp(mag_cat)

                # ⑤ ★追加: mlp_one_add または mlp_one_mul の場合、単語ベクトル（Semantic Anchor）を作用させる
                if self.temporal_emb_type in ['mlp_one_add', 'mlp_one_mul']:
                    exp_anchor_time = self.anchor_time.unsqueeze(0).expand(seq_len, -1)
                    exp_anchor_mag  = self.anchor_mag.unsqueeze(0).expand(seq_len, -1)
                    exp_anchor_dep  = self.anchor_dep.unsqueeze(0).expand(seq_len, -1)

            elif self.temporal_emb_type in ['MLP_add', 'MLP_mul']:

                exp_anchor_time = self.anchor_time.unsqueeze(0).expand(time_vec.size(0), -1)

                # ③ 足し算(Add) または 掛け算(Mul) で結合
                if self.temporal_emb_type == 'MLP_add':

                mag_scalar  = norm_event_mag.unsqueeze(-1)
                dep_scalar  = norm_event_dep.unsqueeze(-1)
                lat_scalar  = norm_event_lat.unsqueeze(-1) # 🌟 追加
                lon_scalar  = norm_event_lon.unsqueeze(-1) # 🌟 追加
                fusion_cat = torch.cat([time_scalar, mag_scalar, dep_scalar, lat_scalar, lon_scalar], dim=-1) # 🌟 5つ結合する
                fusion_vecs=self.fusion_mlp(fusion_cat)

            elif self.temporal_emb_type == 'MLP_fusion_renew':
                time_scalar = norm_event_times.unsqueeze(-1)
                mag_scalar  = norm_event_mag.unsqueeze(-1)

            # 🌟 【追加】MLP_token_renew_loss の処理と特徴量の保持

                time_scalar = norm_event_times.unsqueeze(-1)

                time_scalar = norm_event_times.unsqueeze(-1)

                    r = getattr(self, 'codebook_r_p', torch.tensor(4.6, device=raw_fusion_vecs.device))

                    fusion_vecs = F.normalize(raw_fusion_vecs, p=2, dim=-1) * r

                if self.temporal_emb_type in ['MLP_token_renew_vocab_topk_loss_ph','MLP_token_renew_vocab_loss_ph_scaled','MLP_token_renew_loss', 'MLP_token_renew_loss_p', 'MLP_token_renew_loss_ph','MLP_token_renew_topk_loss_p', 'MLP_token_renew_topk_loss_ph','MLP_token_renew_vocab_loss_ph']:

            # =========================================================================
            
            
            if self.use_tokens:
                if self.use_natural_separators:
                    time_event_text_embedding= self.embed_event_text(time_event_texts)
                    type_event_text_embedding= self.embed_event_text(type_event_texts)

                    dep_event_text_embedding= self.embed_event_text(dep_event_texts)

                    lon_event_text_embedding= self.embed_event_text(lon_event_texts)

                else:
                    time_event_text_embedding= self.embed_event_text(time_event_texts)
                    type_event_text_embedding= self.embed_event_text(type_event_texts)
                    mag_event_text_embedding= self.embed_event_text(mag_event_texts)
                    dep_event_text_embedding= self.embed_event_text(dep_event_texts)
                    lat_event_text_embedding= self.embed_event_text(lat_event_texts)
                    lon_event_text_embedding= self.embed_event_text(lon_event_texts)

            
            event_text_embeddings = self.embed_event_text(event_texts)  # [(text_token_len, embedding_dim), ...]
            
            if self.temporal_emb_type in ['MLP_token_renew_vocab_topk_loss_ph','MLP_token_renew_vocab_loss_ph_scaled','MLP_fusion','MLP_fusion_renew','MLP_token_renew_loss', 'MLP_token_renew_loss_p', 'MLP_token_renew_loss_h', 'MLP_token_renew_loss_ph','MLP_token_renew_topk_loss_p', 'MLP_token_renew_topk_loss_h', 'MLP_token_renew_topk_loss_ph','MLP_token_renew_vocab_loss_ph']:

    def compute_intensities(
        self, batch_event_time_deltas: List[Tensor], batch_hidden_states: List[Tensor],
    ) -> List[Tensor]:
        """
        Compute intensities from event times and hidden states

        :param batch_event_time_deltas: event time deltas in a batch of event sequences, [(seq_len, n_samples), ...]
        :param batch_hidden_states: hidden states for a batch of event sequences, [(seq_len, hidden_size), ...]
        :return: a batch of event intensities, [(seq_len - 1, num_types), ...]
        """
        batch_intensities = []

        for event_time_deltas, hidden_states in zip(batch_event_time_deltas, batch_hidden_states):
            event_time_deltas_tensor = event_time_deltas.unsqueeze(1)[1:]  # (seq_len - 1, 1)
            if self.mlp_flag==True:
                #event_time_deltas_tensor_3d = event_time_deltas_tensor.unsqueeze(-1)
                #print("deltas shape:", event_time_deltas_tensor.shape)  # 例: torch.Size([16, 10])
                #print("hidden shape:", hidden_states[:-1].shape)  # 例: torch.Size([16, 10, 4096])
                mlp_input = torch.cat([event_time_deltas_tensor, hidden_states[:-1]], dim=-1)
                intensity_mlp=self.intensity_MLP(mlp_input).squeeze(-1)
                intensities=self.softplus(intensity_mlp)
            else:
            
                intensities_current = self.intensity_current(event_time_deltas_tensor)  # (seq_len - 1, num_types)
                intensities_history = self.intensity_history(hidden_states[:-1])  # (seq_len - 1, num_types)
                intensities = self.softplus(intensities_current + intensities_history)  # (seq_len - 1, num_types)
                        
            #intensities = self.softplus(intensities_current + intensities_history)  # (seq_len - 1, num_types)
            batch_intensities.append(intensities)  # [(seq_len - 1, num_types), ...]

        return batch_intensities

    def generate_time_deltas(self, batch_event_time_deltas: List[Tensor]) -> List[Tensor]:
        """
        Generate the time delta samples for every interval without padding.

        :param batch_event_time_deltas: list of event times since the last event, [(seq_len,), ...]
        :return: list of time samples for every interval, [(seq_len - 1, n_samples), ...]
        """
        batch_sampled_time_deltas = []

        # Process each sequence in the batch
        for event_time_deltas in batch_event_time_deltas:
            # Convert the tensor of time deltas
            event_time_deltas_tensor = event_time_deltas.unsqueeze(1)[1:]  # (seq_len - 1, 1)

            # Generate the sampling ratios
            time_delta_ratios = torch.linspace(
                start=0.0, end=1.0, steps=self.num_integral_samples, device=self.device)  # (1, n_samples)

            # Sample the time deltas across the intervals
            sampled_time_deltas = event_time_deltas_tensor * time_delta_ratios.unsqueeze(0)  # (seq_len - 1, n_samples)
            batch_sampled_time_deltas.append(sampled_time_deltas)  # [(seq_len - 1, n_samples), ...]

        return batch_sampled_time_deltas

    def compute_sampled_intensities(
        self, batch_sampled_time_deltas: List[Tensor], batch_hidden_states: List[Tensor],
    ) -> List[Tensor]:
        """
        Compute intensities at sampled time deltas in a batch

        :param batch_sampled_time_deltas: a batch of sampled time delta sequence, [(seq_len - 1, n_samples), ...]
        :param batch_hidden_states: a batch of hidden state sequences, [(seq_len, hidden_size), ...]
        :return: a batch of intensities at sampled times, [(seq_len - 1, n_samples, num_types), ...]
        """
        batch_sampled_intensities = []

        for sampled_time_deltas, hidden_states in zip(batch_sampled_time_deltas, batch_hidden_states):
            sampled_time_deltas_tensor = sampled_time_deltas.unsqueeze(-1)  # (seq_len - 1, n_samples, 1)
            
            if self.mlp_flag==True:
                # 🌟 末尾に次元を追加して 2D [B, L] -> 3D [B, L, 1] に変換
                num_samples = sampled_time_deltas_tensor.shape[1]  # 20
                hidden_expanded = hidden_states[:-1].unsqueeze(1)  # [5, 1, 2048]
                # 2. サンプル数(20)に合わせてコピー（メモリを消費しない expand を使用）
                hidden_expanded = hidden_expanded.expand(
                    -1, num_samples, -1
                )  # [5, 20, 2048]
                #print("deltas shape2:", sampled_time_deltas_tensor.shape)  # 例: torch.Size([16, 10])
                #print("hidden shape2:", hidden_expanded.shape)  # 例: torch.Size([16, 10, 4096])
                                
                mlp_input = torch.cat([sampled_time_deltas_tensor, hidden_expanded], dim=-1)
                intensity_mlp=self.intensity_MLP(mlp_input).squeeze(-1)
                sampled_intensities=self.softplus(intensity_mlp)
            else:
                sampled_intensities_current = self.intensity_current(
                    sampled_time_deltas_tensor)  # (seq_len - 1, n_samples, num_types)
                sampled_intensities_history = self.intensity_history(
                    hidden_states[:-1]).unsqueeze(1)  # (seq_len - 1, 1, num_types)
                sampled_intensities = self.softplus(
                    sampled_intensities_current + sampled_intensities_history)  # (seq_len - 1, n_samples, num_types)
            batch_sampled_intensities.append(sampled_intensities)  # [(seq_len - 1, n_samples, num_types), ...]

        return batch_sampled_intensities

    def compute_log_likelihood(
        self, batch_event_time_deltas: List[Tensor], batch_event_types: List[Tensor],
        batch_hidden_states: List[Tensor]) -> List[Tensor]:
        """
        Compute log likelihoods of event sequences in a batch

        :param batch_event_time_deltas: a batch of event time deltas, [(seq_len,), ...]
        :param batch_event_types: a batch of event types, [(seq_len,), ...]
        :param batch_hidden_states: a batch of hidden states, [(seq_len, hidden_size), ...]
        :return: a batch of log likelihoods, [(seq_len - 1,), ...]
        """
        # Compute intensities of events
        batch_event_intensities = self.compute_intensities(
            batch_event_time_deltas=batch_event_time_deltas,
            batch_hidden_states=batch_hidden_states)  # [(seq_len - 1, num_types), ...]

        # Sample time deltas and compute their intensities
        batch_sampled_time_deltas = self.generate_time_deltas(
            batch_event_time_deltas=batch_event_time_deltas)  # [(seq_len - 1, n_samples), ...]
        batch_sampled_intensities = self.compute_sampled_intensities(
            batch_sampled_time_deltas=batch_sampled_time_deltas,
            batch_hidden_states=batch_hidden_states)  # [(seq_len - 1, n_samples, num_types), ...]

        # Compute log likelihoods for the event part
        batch_log_likelihoods = []

        # event_types: (seq_len,), event_time_deltas: (seq_len,)
        # event_intensities: (seq_len - 1, num_types), sampled_intensities: (seq_len - 1, n_samples, num_types)
        for event_types, event_time_deltas, event_intensities, sampled_intensities in zip(
            batch_event_types, batch_event_time_deltas, batch_event_intensities, batch_sampled_intensities):
            # Compute the log likelihood part of events
            event_type_masks = F.one_hot(
                event_types[1:], num_classes=self.num_event_types).to(self.device)  # (seq_len - 1, num_types)
            event_likelihoods = torch.sum(event_intensities * event_type_masks, dim=-1)  # (seq_len - 1,)
            event_log_likelihoods = torch.log(event_likelihoods + 1e-8)  # (seq_len - 1,)

            # Compute the log likelihood part of non-events
            sampled_total_intensities = torch.sum(sampled_intensities, dim=-1)  # (seq_len - 1, n_samples)
            non_event_log_likelihoods = \
                sampled_total_intensities.mean(dim=-1) * event_time_deltas[1:]  # (seq_len - 1,)

            log_likelihoods = event_log_likelihoods - non_event_log_likelihoods  # (seq_len - 1,)
            batch_log_likelihoods.append(log_likelihoods)

        return batch_log_likelihoods

    def predict_next_event_probs(self, batch_hidden_states: List[Tensor]) -> Tuple[List[Tensor], List[Tensor]]:
        """
        Predict next events with probabilities (for training)

        :param batch_hidden_states: hidden states for a batch of event sequences, [(seq_len, hidden_size), ...]
        :return: a batch of next event type probabilities (in [(seq_len, num_types), ...])
            and next event times (in [(seq_len,), ...])
        """
        batch_next_event_type_probs = []
        batch_next_event_times = []

        for hidden_states in batch_hidden_states:
            next_event_type_probs = self.head_type(hidden_states)  # (seq_len, num_types)
            next_event_times = self.head_time(hidden_states).squeeze()  # (seq_len,)
            batch_next_event_type_probs.append(next_event_type_probs)  # [(seq_len, num_types), ...]
            batch_next_event_times.append(next_event_times)  # [(seq_len,), ...]

        return batch_next_event_type_probs, batch_next_event_times

import torch.nn.functional as F


class WeightedKappaLoss(nn.Module):
    
    
    

    def __init__(
        self,
        num_classes: int,
        mode: str = "quadratic",
        epsilon: float = 1e-7,
        device: Union[str, torch.device] = 'cpu'
        
    ):
        super().__init__()
        self.device= device

        if num_classes < 2:
            raise ValueError("num_classes must be >= 2")

        if mode not in ("linear", "quadratic"):
            raise ValueError(
                "mode must be 'linear' or 'quadratic'"
            )

        self.num_classes = num_classes
        self.epsilon = epsilon

        # Ordinal disagreement weights
        classes = torch.arange(num_classes, dtype=torch.float32)
        distances = (classes[:, None] - classes[None, :]).abs()

        if mode == "quadratic":
            weights = distances.square() / (num_classes - 1) ** 2
        else:
            weights = distances / (num_classes - 1)

        self.register_buffer("weights", weights)

    def forward(self, y_pred, y_true):
        # y_pred must be logits, not softmax probabilities
        probs = F.softmax(y_pred, dim=-1)

        y_true = y_true.reshape(-1).long()
        targets = F.one_hot(
            y_true, num_classes=self.num_classes
        ).to(dtype=probs.dtype)

        batch_size = probs.size(0)

        # Soft confusion matrix: rows=predictions, columns=targets
        conf_mat = probs.T @ targets

        # Observed weighted disagreement
        observed = (self.weights * conf_mat).sum()

        # Expected confusion matrix under independence
        hist_pred = probs.sum(dim=0)
        hist_true = targets.sum(dim=0)

        expected_mat = (
            hist_pred[:, None] * hist_true[None, :]
        ) / batch_size

        # Expected weighted disagreement
        expected = (self.weights * expected_mat).sum()

        return observed / expected.clamp_min(self.epsilon)

class TPPLLMRunner(object):
    """
    TPP-LLM Runner
    """

    def __init__(
        self, model: TPPLLMModel, beta_type: float = 1, beta_time: float = 1,device: Union[str, torch.device] = 'cpu'
        ,weight_path:str='',load_flag:bool=False):
        """
        Initialize the TPP-LLM runner.

        :param model: The TPPLLMModel instance.
        :param beta_type: Loss coefficient for event type prediction.
        :param beta_time: Loss coefficient for event time prediction.
        :param beta_semantic: Loss coefficient for semantic alignment.
        :param device: Execution device.
        """


        self.model = model
        self.beta_type = 0.1
        self.beta_time = 1
        self.device = torch.device(device)

        self.num_train_epochs = 0
        self.num_training_steps = 0
        self.num_warmup_steps = 0
        self.global_step = 0
        self.scheduler = None
        self.optimizer = None

        if(load_flag):
            self.load(weight_path)

        # Lambda functions to detach tensors and move them to CPU as numpy arrays
        self.move_to_numpy_prev = lambda list_tensors: [tensor[1:].numpy(force=True) for tensor in list_tensors]
        self.move_to_numpy_next = lambda list_tensors: [tensor[:-1].numpy(force=True) for tensor in list_tensors]

    def run_batch(self, batch: Dict[str, list], phase: str) \
        -> Tuple[np.ndarray, np.ndarray, List[np.ndarray], List[np.ndarray], List[np.ndarray], List[np.ndarray],List[np.ndarray]]:
        """
        Run a single batch of event sequences.

        Returns:
            Tuple containing:
            - event_nums: Number of events per sequence
            - log_likelihoods: Log-likelihoods of the sequences
            - event_types_shifted: True event types (shifted for prediction)
            - next_event_types_shifted: Predicted next event types
            - event_times_shifted: True event times (shifted)
            - next_event_times_shifted: Predicted next event times
            - event_time_deltas_shifted: True time deltas (shifted)
            - semantic_loss: Semantic alignment loss (float)
            - batch_loss: Total combined loss (Tensor, for backward pass)
            - event_mags_shifted: True event magnitudes (shifted)
        """

        # Move inputs to the target device
        batch = {
            'time_since_start': [_seq.to(self.device) for _seq in batch['time_since_start']],
            'time_since_last_event': [_seq.to(self.device) for _seq in batch['time_since_last_event']],
            'type_event': [_seq.to(self.device) for _seq in batch['type_event']],
            'type_text':batch['type_text'],
            'magnitude':[_seq.to(self.device) for _seq in batch['magnitude']],
            'depth':[_seq.to(self.device) for _seq in batch['depth']],
            'latitude':[_seq.to(self.device) for _seq in batch['latitude']],
            'longitude':[_seq.to(self.device) for _seq in batch['longitude']],
            'time_float':batch['time_float'],
            'magnitude_float':batch['magnitude_float'],
            'depth_float':batch['depth_float'],
            'latitude_float':batch['latitude_float'],
            'longitude_float':batch['longitude_float'],
        }

        if phase == 'train':
            self.model.train()

            

            # ========== 変更後 ==========
            # 🌟 13個の値を受け取る
            #batch_event_nums, batch_nll_losses, batch_type_losses, batch_time_losses, batch_next_event_types, batch_next_event_times, codebook_losses, batch_words_p, batch_words_h, batch_distination_p, batch_distination_h, batch_dist_eq_p, batch_dist_eq_h = self.model.compute_loss(batch)
            batch_event_nums, batch_nll_losses, batch_type_losses, batch_time_losses, batch_next_event_types, batch_next_event_times, codebook_losses, batch_words_p, batch_words_h, batch_distination_p, batch_distination_h, batch_dist_eq_p, batch_dist_eq_h, raw_cb_p, raw_cb_h ,temp_embs_for_loss,batch_hidden_states= self.model.compute_loss(batch)
            # Lossの計算をシンプルに
            #batch_loss = torch.sum(
            #    batch_nll_losses + self.beta_type * batch_type_losses + self.beta_time * batch_time_losses)
            # 1. 基本のタスク損失（系列ごとの配列）
            base_losses = 0.3*batch_nll_losses + self.beta_type * batch_type_losses + self.beta_time * batch_time_losses
            batch_type_loss = torch.sum(batch_type_losses)
            batch_time_loss = torch.sum(batch_time_losses)
            batch_nll_loss = torch.sum(batch_nll_losses)
            # 2. 🌟 Codebook loss が存在する場合は重みをかけて加算する
            if codebook_losses is not None:
                total_losses_per_seq = base_losses +  codebook_losses
                # 最後に全体を合計する
                batch_loss = torch.sum(total_losses_per_seq)
                
                # ログ・コンソール出力用のスカラー値（ペナルティ適用後の合計値）
                loss_codebook_val = torch.sum( codebook_losses).item()
            else:
                batch_loss = torch.sum(base_losses)
                loss_codebook_val = 0.0
        
            batch_event_nums = batch_event_nums.numpy(force=True)
            batch_log_likelihoods = - batch_nll_losses.numpy(force=True)
            

            metrics = {
                'batch_loss': batch_loss.float().cpu().item(),
                'loss_tpp': torch.sum(batch_nll_losses).float().cpu().item(),
                'loss_type': torch.sum(batch_type_losses).float().cpu().item(),
                'loss_time': torch.sum(batch_time_losses).float().cpu().item(),
                # semantic_loss は削除
                'loss_codebook': loss_codebook_val,  # 🌟 コンソール出力・ログに追加
                'raw_P_loss': torch.sum(raw_cb_p).item() if raw_cb_p is not None else 0.0, # 🌟 コンソール用
                'raw_H_loss': torch.sum(raw_cb_h).item() if raw_cb_h is not None else 0.0, # 🌟 コンソール用
                'batch_event_nums': batch_event_nums.sum().item(),
                'batch_log_likelihood': batch_log_likelihoods.sum().item() / batch_event_nums.sum().item(),
                'learning_rate': self.optimizer.param_groups[0]['lr'],
                'epoch': self.global_step / self.num_training_steps * self.num_train_epochs,
            }
            print(metrics)
            # 🌟 ここを変更: Trainでも予測結果をNumpy配列に変換して返す
            batch_event_types_shifted = self.move_to_numpy_prev(batch['type_event'])
            batch_event_times_shifted = self.move_to_numpy_prev(batch['time_since_start'])
            batch_next_event_types_shifted = self.move_to_numpy_next(batch_next_event_types)
            batch_next_event_times_shifted = self.move_to_numpy_next(batch_next_event_times)
            batch_event_time_deltas_shifted = self.move_to_numpy_prev(batch['time_since_last_event'])

            # 戻り値を11個に増やす（15個受け取って必要なものを返す）
            return batch_event_nums, batch_log_likelihoods, batch_event_types_shifted, \
                   batch_next_event_types_shifted, batch_event_times_shifted, \
                   batch_next_event_times_shifted, batch_event_time_deltas_shifted, \
                   batch_loss, batch_words_p, batch_words_h, batch_distination_p, batch_distination_h,batch_dist_eq_p, batch_dist_eq_h, raw_cb_p, raw_cb_h,temp_embs_for_loss,batch_hidden_states,batch_type_loss ,batch_time_loss,batch_nll_loss

        elif phase == 'eval':
            self.model.eval()

            batch_event_times = batch['time_since_start']
            batch_event_types = batch['type_event']
            batch_event_time_deltas = batch['time_since_last_event']

            # ========== 変更後 ==========
            with torch.no_grad():
                # 🌟 compute_loss 1回で「Loss」も「予測結果」も一気に取得する！
                # 🌟 13個の値を受け取る
                #batch_event_nums, batch_nll_losses, batch_type_losses, batch_time_losses, batch_next_event_types, batch_next_event_times, codebook_losses, batch_words_p, batch_words_h, batch_distination_p, batch_distination_h, batch_dist_eq_p, batch_dist_eq_h = self.model.compute_loss(batch)
                batch_event_nums, batch_nll_losses, batch_type_losses, batch_time_losses, batch_next_event_types, batch_next_event_times, codebook_losses, batch_words_p, batch_words_h, batch_distination_p, batch_distination_h, batch_dist_eq_p, batch_dist_eq_h, raw_cb_p, raw_cb_h ,temp_embs_for_loss,batch_hidden_states= self.model.compute_loss(batch)
                #batch_loss = torch.sum(batch_nll_losses + self.beta_type * batch_type_losses + self.beta_time * batch_time_losses).float().cpu().item()
                # 2. 🌟 学習時と同様に codebook_losses があれば重みをかけて加算する
                # 1. 基本のタスク損失を計算
                #base_losses = batch_nll_losses + self.beta_type * batch_type_losses + self.beta_time * batch_time_losses
                #if codebook_losses is not None:
                #    total_losses = base_losses + self.model.codebook_loss_weight * codebook_losses
                #else:
                #    total_losses = base_losses

                base_losses = 0.3*batch_nll_losses + self.beta_type * batch_type_losses + self.beta_time * batch_time_losses

                # シーケンス全体の合計を出してスカラーに変換
                batch_loss = torch.sum(base_losses).float().cpu().item()
                batch_type_loss = torch.sum(batch_type_losses).float().cpu().item()
                batch_time_loss = torch.sum(batch_time_losses).float().cpu().item()
                batch_nll_loss = torch.sum(batch_nll_losses).float().cpu().item()
                
                batch_log_likelihoods = -batch_nll_losses
                
                # 3. シーケンス全体の合計を出してスカラーに変換
                #batch_loss = torch.sum(total_losses).float().cpu().item()
                #batch_log_likelihoods = -batch_nll_losses
            
            # Predict next events
            #batch_event_nums, batch_log_likelihoods, batch_next_event_types, batch_next_event_times = \
            #    self.model.predict_next_events(batch)

            # Shift and convert tensors
            batch_event_nums = batch_event_nums.numpy(force=True)
            batch_log_likelihoods = batch_log_likelihoods.numpy(force=True)
            batch_event_types_shifted = self.move_to_numpy_prev(batch_event_types)
            batch_event_times_shifted = self.move_to_numpy_prev(batch_event_times)
            batch_next_event_types_shifted = self.move_to_numpy_next(batch_next_event_types)
            batch_next_event_times_shifted = self.move_to_numpy_next(batch_next_event_times)
            batch_event_time_deltas_shifted = self.move_to_numpy_prev(batch_event_time_deltas)
            
            # 戻り値を変更 
            return batch_event_nums, batch_log_likelihoods, batch_event_types_shifted, \
                batch_next_event_types_shifted, batch_event_times_shifted, \
                batch_next_event_times_shifted, batch_event_time_deltas_shifted, \
                batch_loss, batch_words_p, batch_words_h, batch_distination_p, batch_distination_h,batch_dist_eq_p, batch_dist_eq_h, raw_cb_p, raw_cb_h,temp_embs_for_loss,batch_hidden_states,batch_type_loss,batch_time_loss,batch_nll_loss

        else:
            raise KeyError(f'Unknown phase: {phase}.')

    def run_epoch(self, dataloader: DataLoader, phase: str, result_save_path:str, gradient_accumulation_steps: int = 1) -> dict:
        """
        Run an epoch (Updated for Gradient Accumulation)
        :param gradient_accumulation_steps: 勾配蓄積数 (デフォルト1)
        """

        total_log_likelihood = 0
        total_num_events = 0
        total_val_loss = 0.0  # 🌟 ここに追加 (total_semantic_loss は消す)
        total_train_loss = 0.0  # 🌟 追加 (Train時のLoss集計用)
        total_val_type_loss = 0.0  # 🌟 ここに追加 (total_semantic_loss は消す)
        total_val_time_loss = 0.0  # 🌟 ここに追加 (total_semantic_loss は消す)
        total_val_nll_loss = 0.0  # 🌟 ここに追加 (total_semantic_loss は消す)
        total_train_type_loss = 0.0  # 🌟 追加 (Train時のLoss集計用)
        total_train_time_loss = 0.0  # 🌟 追加 (Train時のLoss集計用)
        total_train_nll_loss = 0.0  # 🌟 追加 (Train時のLoss集計用)
        total_raw_p_loss = 0.0  # 🌟 追加 (生のP層ロス集計用)
        total_raw_h_loss = 0.0  # 🌟 追加 (生のH層ロス集計用)
        num_batches = 0
        
        raw_event_types = []
        raw_event_type_preds = []
        raw_event_times = []
        raw_event_time_preds = []
        raw_event_time_deltas = []
        
        all_event_types = []
        all_event_type_preds = []
        all_event_times = []
        all_event_time_preds = []
        all_event_time_deltas = []
        
        raw_event_data_list = []
        all_seq_ll = []  # 🌟 追加：系列ごとの対数尤度を保存
        all_inf_times = []
        metrics = {}

        # ★ Trainフェーズ開始時に勾配を初期化
        if phase == 'train':
            self.optimizer.zero_grad()

        # ★ enumerate でステップ数を取得
        total_inference_time = 0.0  # ⭕️ 追加: 推論時間の合計用変数

        # ★ enumerate でステップ数を取得
        for step, batch in enumerate(tqdm(dataloader)):

            # ⭕️ 追加: evalフェーズのみ時間計測を開始
            
            start_inf_time = time.time()
            
            # 🌟 14個の値を受け取る
            #event_nums, log_likelihoods, event_types, event_type_preds, event_times, \
            #event_time_preds, event_time_deltas, batch_loss_tensor, batch_words_p, batch_words_h, batch_distination_p, batch_distination_h, batch_dist_eq_p, batch_dist_eq_h = \
            #self.run_batch(batch=batch, phase=phase)

            # 🌟 16個の値を受け取る
            event_nums, log_likelihoods, event_types, event_type_preds, event_times, \
            event_time_preds, event_time_deltas, batch_loss_tensor, batch_words_p, batch_words_h, batch_distination_p, batch_distination_h, batch_dist_eq_p, batch_dist_eq_h, raw_cb_p, raw_cb_h ,temp_embs_for_loss,batch_hidden_states,batch_type_loss,batch_time_loss,batch_nll_loss= \
            self.run_batch(batch=batch, phase=phase)

            batch_inf_time = time.time() - start_inf_time
            total_inference_time += batch_inf_time

            # 🌟 生ロスの集計 (テンソルが返ってきている場合)
            if raw_cb_p is not None:
                total_raw_p_loss += float(torch.sum(raw_cb_p).cpu().item())
            if raw_cb_h is not None:
                total_raw_h_loss += float(torch.sum(raw_cb_h).cpu().item())

            # ⭕️ 追加: 1バッチの推論が終わったら時間を足す
            if phase == 'eval':
                #total_inference_time += (time.time() - start_inf_time)
                total_val_loss += batch_loss_tensor # 🌟 追加 (Val_lossの集計)
                total_val_type_loss += batch_type_loss # 🌟 追加 (Val_lossの集計)
                total_val_time_loss += batch_time_loss # 🌟 追加 (Val_lossの集計)
                total_val_nll_loss += batch_nll_loss # 🌟 追加 (Val_lossの集計)
                # 🌟 追加: 推論時間のばらつき（標準偏差）を記録
            batch_events = np.sum(event_nums)
            if batch_events > 0:
                time_per_event_ms = (batch_inf_time / batch_events) * 1000
                # バッチ内の系列すべてに、この平均処理時間を記録
                all_inf_times.append(time_per_event_ms)
                
            
            #total_log_likelihood += np.sum(log_likelihoods)
            #total_num_events += np.sum(event_nums)

            # 変更後
            total_log_likelihood += np.sum(log_likelihoods)
            total_num_events += np.sum(event_nums)
            
            # 🌟 追加：系列ごとの対数尤度を保存
            if len(event_nums) > 0:
                seq_lls = log_likelihoods / event_nums
                all_seq_ll.extend(seq_lls.tolist())
            
            if phase == 'train':
                num_batches += 1  # total_semantic_loss += semantic_loss は削除
                total_train_loss += float(batch_loss_tensor) # 🌟 追加 (floatで値だけ取り出して足すことでメモリリーク防止)
                total_train_type_loss += float(batch_type_loss)
                total_train_time_loss += float(batch_time_loss)
                total_train_nll_loss += float(batch_nll_loss)
                # 1. Average the loss over the accumulation steps
                loss = batch_loss_tensor / gradient_accumulation_steps
                
                # 2. Backward pass (accumulate gradients)
                loss.backward()
                
                # 3. Update parameters if we reached the accumulation step limit or the end of the dataloader
                if (step + 1) % gradient_accumulation_steps == 0 or (step + 1) == len(dataloader):
                    
                    # Zero out the gradients of the original LLM embeddings to keep them frozen
                    emb = self.model.llm.get_input_embeddings()
                    if emb.weight.grad is not None:
                        emb.weight.grad[:self.model.old_vocab_size].zero_()

                    if emb.weight.grad is not None:
                        nonzero = (emb.weight.grad.abs().sum(dim=1) > 0).sum().item()
                        print(f"[grad-check] nonzero embedding rows: {nonzero} (should equal num_added_tokens)")
            

                    
                    # Update the parameters
                    self.optimizer.step()
                    
                    # Update the learning rate
                    self.scheduler.step()
                    
                    # Optimize the model parameters
                    self.optimizer.zero_grad()
                    
                    # Update the Global Step
                    self.global_step += 1

                    
            
                    

                # ========================================================

            
            

            # 🌟 追加: オリジナルのバッチから補助情報をNumpyとして取り出しておく
            batch_mags = [m.cpu().numpy() for m in batch['magnitude']]
            batch_deps = [d.cpu().numpy() for d in batch['depth']]
            batch_lats = [l.cpu().numpy() for l in batch['latitude']]
            batch_lons = [l.cpu().numpy() for l in batch['longitude']]
            batch_event_time_t= [l.cpu().numpy() for l in batch['time_since_start']]
            batch_event_type_t= [l.cpu().numpy() for l in batch['type_event']]
            # runner.py 内
            batch_dist_eq_p = locals().get('batch_dist_eq_p', [])
            batch_dist_eq_h = locals().get('batch_dist_eq_h', [])
            
            
            # ============================================================
            # raw_data 保存
            # ============================================================
            if phase == 'eval':
            
                for i in range(len(event_types)):

                    # --------------------------------------------
                    # Input vector
                    # --------------------------------------------
                    if (
                        temp_embs_for_loss
                        and i < len(temp_embs_for_loss)
                    ):
                        input_vector = (
                            temp_embs_for_loss[i]
                            .numpy()
                            .tolist()
                        )
                    else:
                        input_vector = []

                    # --------------------------------------------
                    # Hidden vector
                    # --------------------------------------------
                    if (
                        batch_hidden_states
                        and i < len(batch_hidden_states)
                    ):
                        hidden_vector = (
                            batch_hidden_states[i]
                            .numpy()
                            .tolist()
                        )
                    else:
                        hidden_vector = []

                    # --------------------------------------------
                    # raw_data
                    # --------------------------------------------
                    raw_event_data_list.append({

                        'true_types': (
                            event_types[i].tolist()
                            if isinstance(
                                event_types[i],
                                np.ndarray
                            )
                            else event_types[i]
                        ),

                        'pred_types': (
                            event_type_preds[i].tolist()
                            if isinstance(
                                event_type_preds[i],
                                np.ndarray
                            )
                            else event_type_preds[i]
                        ),

                        'true_times': (
                            event_times[i].tolist()
                            if isinstance(
                                event_times[i],
                                np.ndarray
                            )
                            else event_times[i]
                        ),

                        'pred_times': (
                            event_time_preds[i].tolist()
                            if isinstance(
                                event_time_preds[i],
                                np.ndarray
                            )
                            else event_time_preds[i]
                        ),

                        'mags': (
                            batch_mags[i][0:].tolist()
                            if i < len(batch_mags)
                            else []
                        ),

                        'deps': (
                            batch_deps[i][0:].tolist()
                            if i < len(batch_deps)
                            else []
                        ),

                        'lats': (
                            batch_lats[i][0:].tolist()
                            if i < len(batch_lats)
                            else []
                        ),

                        'lons': (
                            batch_lons[i][0:].tolist()
                            if i < len(batch_lons)
                            else []
                        ),

                        'time': (
                            batch_event_time_t[i][0:].tolist()
                            if i < len(batch_event_time_t)
                            else []
                        ),

                        'type': (
                            batch_event_type_t[i][0:].tolist()
                            if i < len(batch_event_type_t)
                            else []
                        ),

                        'tsne_type': (
                            batch_event_type_t[i].tolist()
                            if i < len(batch_event_type_t)
                            else []
                        ),

                        'tsne_time': (
                            batch_event_time_t[i].tolist()
                            if i < len(batch_event_time_t)
                            else []
                        ),

                        'words_p': (
                            batch_words_p[i][0:]
                            if (
                                batch_words_p
                                and i < len(batch_words_p)
                            )
                            else []
                        ),

                        'words_h': (
                            batch_words_h[i][0:]
                            if (
                                batch_words_h
                                and i < len(batch_words_h)
                            )
                            else []
                        ),

                        'dist_p': (
                            batch_distination_p[i][0:]
                            if (
                                batch_distination_p
                                and i < len(batch_distination_p)
                            )
                            else []
                        ),

                        'dist_h': (
                            batch_distination_h[i][0:]
                            if (
                                batch_distination_h
                                and i < len(batch_distination_h)
                            )
                            else []
                        ),

                        'dist_eq_p': (
                            batch_dist_eq_p[i][0:]
                            if (
                                batch_dist_eq_p
                                and i < len(batch_dist_eq_p)
                            )
                            else []
                        ),

                        'dist_eq_h': (
                            batch_dist_eq_h[i][0:]
                            if (
                                batch_dist_eq_h
                                and i < len(batch_dist_eq_h)
                            )
                            else []
                        ),

                        # ====================================================
                        # ★ t-SNE 用 Input / Hidden vector
                        # ====================================================
                        'input_vector': input_vector,

                        'hidden_vector': hidden_vector,
                    })
                """
                # 🌟🌟🌟 追加: 補助情報と選ばれた単語も保存 (他のデータと長さを合わせるため [1:] でスライス)
            for i in range(len(event_types)):
                raw_event_data_list.append({
                    'true_types': event_types[i].tolist() if isinstance(event_types[i], np.ndarray) else event_types[i],
                    'pred_types': event_type_preds[i].tolist() if isinstance(event_type_preds[i], np.ndarray) else event_type_preds[i],
                    'true_times': event_times[i].tolist() if isinstance(event_times[i], np.ndarray) else event_times[i],
                    'pred_times': event_time_preds[i].tolist() if isinstance(event_time_preds[i], np.ndarray) else event_time_preds[i],
                    
                    'mags': batch_mags[i][0:].tolist() if i < len(batch_mags) else [],
                    'deps': batch_deps[i][0:].tolist() if i < len(batch_deps) else [],
                    'lats': batch_lats[i][0:].tolist() if i < len(batch_lats) else [],
                    'lons': batch_lons[i][0:].tolist() if i < len(batch_lons) else [],
                    'time': batch_event_time_t[i][0:].tolist() if i < len(batch_event_time_t) else [],
                    'type': batch_event_type_t[i][0:].tolist() if i < len(batch_event_type_t) else [],
                    'tsne_type': batch_event_type_t[i].tolist() if i < len(batch_event_type_t) else [],
                    'tsne_time': batch_event_time_t[i].tolist() if i < len(batch_event_time_t) else [],
                    'words_p': batch_words_p[i][0:] if batch_words_p and i < len(batch_words_p) else [],
                    'words_h': batch_words_h[i][0:] if batch_words_h and i < len(batch_words_h) else [],
                    'dist_p': batch_distination_p[i][0:] if batch_distination_p and i < len(batch_distination_p) else [],
                    'dist_h': batch_distination_h[i][0:] if batch_distination_h and i < len(batch_distination_h) else [],
                    'dist_eq_p': batch_dist_eq_p[i][0:] if batch_dist_eq_p and i < len(batch_dist_eq_p) else [],
                    'dist_eq_h': batch_dist_eq_h[i][0:] if batch_dist_eq_h and i < len(batch_dist_eq_h) else [],
                    'input_vector':temp_embs_for_loss[i] if temp_embs_for_loss and i < len(temp_embs_for_loss) else [],
                    'hidden_vector':batch_hidden_states[i] if batch_hidden_states and i < len(batch_hidden_states) else [],


                })
            """
            all_event_types.append(np.concatenate(event_types))
            all_event_type_preds.append(np.concatenate(event_type_preds))
            all_event_times.append(np.concatenate(event_times))
            all_event_time_preds.append(np.concatenate(event_time_preds))
            all_event_time_deltas.append(np.concatenate(event_time_deltas))

            raw_event_types.extend(event_types)
            raw_event_type_preds.extend(event_type_preds)
            raw_event_times.extend(event_times)
            raw_event_time_preds.extend(event_time_preds)
            raw_event_time_deltas.extend(event_time_deltas)

        # Calculate final metrics for the epoch
        avg_log_likelihood = total_log_likelihood / total_num_events
        metrics['log_likelihood'] = float(avg_log_likelihood)
        metrics['num_events'] = int(total_num_events)
        
        if phase == 'train' and num_batches > 0:
            # 🌟 追加 (全イベントで割って、1イベントあたりの平均Train Lossを算出)
            metrics['train_loss'] = float(total_train_loss / total_num_events) if total_num_events > 0 else 0.0
            metrics['train_type_loss'] = float(total_train_type_loss / total_num_events) if total_num_events > 0 else 0.0
            metrics['train_time_loss'] = float(total_train_time_loss / total_num_events) if total_num_events > 0 else 0.0
            metrics['train_nll_loss'] = float(total_train_nll_loss / total_num_events) if total_num_events > 0 else 0.0
        elif phase == 'eval':
            metrics['val_loss'] = float(total_val_loss / total_num_events) if total_num_events > 0 else 0.0
            metrics['val_type_loss'] = float(total_val_type_loss / total_num_events) if total_num_events > 0 else 0.0
            metrics['val_time_loss'] = float(total_val_time_loss / total_num_events) if total_num_events > 0 else 0.0
            metrics['val_nll_loss'] = float(total_val_nll_loss / total_num_events) if total_num_events > 0 else 0.0
        # 🌟 生のLossの平均も epoch 全体の metrics に保存する (Train/Eval/Test 共通)
        metrics['raw_P_loss'] = float(total_raw_p_loss / total_num_events) if total_num_events > 0 else 0.0
        metrics['raw_H_loss'] = float(total_raw_h_loss / total_num_events) if total_num_events > 0 else 0.0
        
        all_event_types = np.concatenate(all_event_types)
        all_event_type_preds = np.concatenate(all_event_type_preds)
        all_event_times = np.concatenate(all_event_times)
        all_event_time_preds = np.concatenate(all_event_time_preds)
        all_event_time_deltas = np.concatenate(all_event_time_deltas)

        accuracy = np.mean(all_event_types == all_event_type_preds)
        rmse = np.sqrt(np.mean((all_event_times - all_event_time_preds) ** 2))

            # ★ ここに追加
        precision, recall, f1, _ = precision_recall_fscore_support(
            all_event_types, 
            all_event_type_preds, 
            labels=[0,1,2],
            average='macro', 
            zero_division=0 # クラスによっては予測がない場合にエラーを防ぐため
        )
        from collections import Counter

        print("True class counts:", Counter(all_event_types))
        print("Pred class counts:", Counter(all_event_type_preds))
        print("Confusion matrix:")
        from sklearn.metrics import confusion_matrix
        print(confusion_matrix(
            all_event_types,
            all_event_type_preds,
            labels=[0, 1, 2]
        ))
        # 全イベントの Quadratic Weighted Kappa
        qwk = cohen_kappa_score(
            all_event_types,
            all_event_type_preds,
            labels=[0, 1, 2],
            weights='quadratic'
        )
            
        metrics['accuracy'] = float(accuracy)
        metrics['rmse'] = float(rmse)
        metrics['precision'] = float(precision) # 追加
        metrics['recall'] = float(recall)       # 追加
        metrics['f1'] = float(f1)               # 追加
        metrics['qwk'] = float(qwk)
            # ⭕️ 追加: 1イベントあたりの平均推論時間をミリ秒（ms）で計算
        avg_inf_time_ms = (total_inference_time / total_num_events) * 1000 if total_num_events > 0 else 0
        metrics['avg_inf_time_per_event(ms)'] = float(avg_inf_time_ms)
            
            
        seq_accuracies = []
        seq_rmses = []
        seq_f1s = []
        seq_qwks = []

        for true_typ, pred_typ, true_tm, pred_tm in zip(
            raw_event_types, raw_event_type_preds, raw_event_times, raw_event_time_preds
        ):
            seq_acc = np.mean(true_typ == pred_typ)
            seq_accuracies.append(float(seq_acc))
            seq_rmse = np.sqrt(np.mean((true_tm - pred_tm) ** 2))
            seq_rmses.append(float(seq_rmse))
            # 系列ごとの F1スコア を計算
            _, _, seq_f1, _ = precision_recall_fscore_support(true_typ, pred_typ,labels=[0,1,2], average='macro', zero_division=0)
            seq_f1s.append(float(seq_f1))
            # 系列ごとの QWK
            seq_qwk = cohen_kappa_score(
                true_typ,
                pred_typ,
                labels=[0, 1, 2],
                weights='quadratic'
            )
            seq_qwks.append(float(seq_qwk))
            
        # 辞書に標準偏差を登録
        metrics['log_likelihood_std'] = float(np.std(all_seq_ll)) if len(all_seq_ll) > 0 else 0.0
        metrics['rmse_std'] = float(np.std(seq_rmses)) if len(seq_rmses) > 0 else 0.0
        metrics['f1_std'] = float(np.std(seq_f1s)) if len(seq_f1s) > 0 else 0.0
        metrics['qwk_std'] = (
            float(np.nanstd(seq_qwks))
            if np.any(np.isfinite(seq_qwks)) else 0.0
        )
        metrics['inf_time_std(ms)'] = float(np.std(all_inf_times)) if len(all_inf_times) > 0 else 0.0
        # ====================================================

        seq_scores = {
            "accuracies": seq_accuracies,
           "rmses": seq_rmses,
            "f1s": seq_f1s, # これも追加しておくと後で分析しやすいです
            "lls": all_seq_ll,  # 🌟 この1行を追加
            "raw_data": raw_event_data_list
        }
        # 🌟 呼び出し元（runメソッド）の形を崩さないよう、evalの時だけ tuple を返す
        if phase == 'eval':
            return metrics, (all_event_types, all_event_type_preds,
                             all_event_times, all_event_time_preds,
                             all_event_time_deltas, seq_scores)

        # train の時は metrics のみを返す
        return metrics

    def run(
        self, quantitative_analysis_func: callable,
        dataloader_train: DataLoader = None, dataloader_val: DataLoader = None,
        dataloader_test: DataLoader = None, learning_rate: float = 5e-4, lr_scheduler_type: str = 'constant',
        num_train_epochs: int = 1, warmup_ratio: float = 0, result_save_path: str = '',
        weight_path: str = '', model_weight_path: str = '', figure_path: str = '', save_flag: bool = False, train_flag: bool = False,resume_flag=False,
        gradient_accumulation_steps: int = 1,temporal_emb_type: str=""
    ) -> None:
        """
        Run the training (Updated)
        """
        # Calculate steps considering accumulation
        self.num_train_epochs = num_train_epochs
        print(gradient_accumulation_steps)
        steps_per_epoch = len(dataloader_train) // gradient_accumulation_steps
        self.num_training_steps = self.num_train_epochs * steps_per_epoch
        self.num_warmup_steps = int(warmup_ratio * self.num_training_steps)
        self.global_step = 0
        self.temporal_emb_type=temporal_emb_type
        
        self.optimizer = Adam(
            list(self.model.parameters()) ,
            lr=learning_rate
        )

        if lr_scheduler_type == 'constant':
            self.scheduler = transformers.get_constant_schedule(optimizer=self.optimizer)
        elif lr_scheduler_type == 'constant_with_warmup':
            self.scheduler = transformers.get_constant_schedule_with_warmup(
                optimizer=self.optimizer, num_warmup_steps=self.num_warmup_steps)
        elif lr_scheduler_type == 'linear':
            self.scheduler = transformers.get_linear_schedule_with_warmup(
                optimizer=self.optimizer, num_warmup_steps=self.num_warmup_steps, num_training_steps=self.num_training_steps)
        elif lr_scheduler_type == 'cosine':
            self.scheduler = transformers.get_cosine_schedule_with_warmup(
                optimizer=self.optimizer, num_warmup_steps=self.num_warmup_steps, num_training_steps=self.num_training_steps, num_cycles=0.5)
        else:
            raise KeyError(f'Unknown learning rate scheduler type: {lr_scheduler_type}')

        metrics_val_best = None

        start_epoch = 0

        if resume_flag:
            base_weight = os.path.join(weight_path, 'model.pth')
            checkpoint_path = os.path.join(weight_path, 'training_checkpoint.pth')
            
            if os.path.exists(base_weight):
                # 保存先パスを「_resume」に変更（既存の動作を維持）
                result_save_path = result_save_path.rstrip('/') + '_resume'
                weight_path = weight_path.rstrip('/') + '_resume'
                model_weight_path = model_weight_path.rstrip('/') + '_resume'
                os.makedirs(result_save_path, exist_ok=True)
                os.makedirs(weight_path, exist_ok=True)
                os.makedirs(model_weight_path, exist_ok=True)

                self.model.load_state_dict(torch.load(base_weight, map_location=self.device, weights_only=False), strict=False)

                # ⬇️ 追加: チェックポイントが存在すれば勢いや乱数を完全に復元
                if os.path.exists(checkpoint_path):
                    checkpoint = torch.load(checkpoint_path, map_location=self.device, weights_only=False)
                    start_epoch = checkpoint['epoch'] + 1
                    
                    if self.optimizer and checkpoint['optimizer_state_dict']:
                        self.optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
                    if self.scheduler and checkpoint['scheduler_state_dict']:
                        self.scheduler.load_state_dict(checkpoint['scheduler_state_dict'])
                    
                    if checkpoint['metrics_val_best']:
                        metrics_val_best = checkpoint['metrics_val_best']
                    
                    # 乱数状態の復元
                    torch.set_rng_state(checkpoint['rng_state'].cpu())
                    # cuda_rng_state はリスト形式で保存されているため、リスト内包表記で全て .cpu() にする
                    torch.cuda.set_rng_state_all([s.cpu() for s in checkpoint['cuda_rng_state']])
                    np.random.set_state(checkpoint['np_rng_state'])
                    random.setstate(checkpoint['py_rng_state'])
                    
                    print(f"\n==================================================")
                    print(f" 🔄 【完全再開モード】 モデル重み、オプティマイザの勢い、乱数状態を復元しました。")
                    print(f" ▶️ Epoch {start_epoch} から学習を継続します。")
                    print(f"==================================================\n")
                else:
                    # 以前のバージョン（チェックポイント未作成）から再開する場合のフォールバック
                    print(f"⚠️ チェックポイントが見つかりません。重みのみをロードします。")
                    # (必要であれば元の val.txt からエポックを逆算する処理をここに残してもOKです)


        # Initial validation
        if dataloader_val:
            # 🌟🌟🌟 修正：run_epochを実行する【前】にリセットとヘッダー書き込みを行う 🌟🌟🌟
            if hasattr(self.model, 'reset_topk_debug_flags'):
                self.model.reset_topk_debug_flags()
                self.model.result_save_path = result_save_path  # モデルに保存先パスを教える
                try:
                    log_file_path = f"{result_save_path}/topk_debug_log.txt"
                    with open(log_file_path, "a", encoding="utf-8") as f:
                        f.write(f"\n{'='*50}\n 🚀 Initial の 検証 Top-k 単語観測 (Validation)\n{'='*50}\n")
                except Exception:
                    pass
            metrics_val, eval_data = self.run_epoch(dataloader_val, phase='eval', result_save_path=result_save_path)
            print(f'validation metrics: {metrics_val}')
            metrics_val_best = metrics_val
            with open(result_save_path+'/val.txt', 'a') as f:
                f.write(f'validation metrics: {metrics_val}\n')
            
            
            (all_types_true, all_types_pred, all_times_true, all_time_preds, all_time_deltas_true, seq_scores_val) = eval_data
            if self.temporal_emb_type in ['MLP_token_renew_vocab_topk_loss_ph','MLP_token_renew_vocab_loss_ph_scaled','MLP_token_renew_loss', 'MLP_token_renew_loss_p', 'MLP_token_renew_loss_h','MLP_token_renew_loss_ph','MLP_token_renew_topk_loss_p', 'MLP_token_renew_topk_loss_h', 'MLP_token_renew_topk_loss_ph','MLP_token_renew_vocab_loss_ph']:
            
                quantitative_analysis_func(
                    all_types_true, all_types_pred, all_times_true, all_time_preds, all_time_deltas_true,
                    result_save_path, "initial_val","initial",self.model, seq_scores_val
                )
            
        # ---------------------------------------------------------
        # Initial Test
        # ---------------------------------------------------------
        if dataloader_test:
            # 🌟🌟🌟 修正：run_epochを実行する【前】にリセットとヘッダー書き込みを行う 🌟🌟🌟
            if hasattr(self.model, 'reset_topk_debug_flags'):
                self.model.reset_topk_debug_flags()
                self.model.result_save_path = result_save_path  # モデルに保存先パスを教える
                try:
                    log_file_path = f"{result_save_path}/topk_debug_log.txt"
                    with open(log_file_path, "a", encoding="utf-8") as f:
                        f.write(f"\n{'='*50}\n 🚀 Initial の テスト Top-k 単語観測 (Validation)\n{'='*50}\n")
                except Exception:
                    pass
            metrics_test, test_eval_data = self.run_epoch(dataloader_test, phase='eval', result_save_path=result_save_path)
            print(f'test metrics: {metrics_test}')
            with open(result_save_path+'/test.txt', 'a') as f:
                f.write(f'test metrics: {metrics_test}\n')

            (all_types_true_test, all_types_pred_test, all_times_true_test, all_time_preds_test, all_time_deltas_true_test, seq_scores_test) = test_eval_data
            if self.temporal_emb_type in ['MLP_token_renew_vocab_topk_loss_ph','MLP_token_renew_vocab_loss_ph_scaled','MLP_token_renew_loss', 'MLP_token_renew_loss_p', 'MLP_token_renew_loss_h','MLP_token_renew_loss_ph','MLP_token_renew_topk_loss_p', 'MLP_token_renew_topk_loss_h', 'MLP_token_renew_topk_loss_ph','MLP_token_renew_vocab_loss_ph']:
                        
                quantitative_analysis_func(
                    all_types_true_test, all_types_pred_test, all_times_true_test, all_time_preds_test, all_time_deltas_true_test,
                    result_save_path, "initial_test","initial", self.model,seq_scores_test
                )
        
        patience = 10
        epochs_no_improve = 0

        

        # ループを start_epoch から始まるように変更
        for epoch in range(start_epoch, int(num_train_epochs)):

            
            if save_flag or train_flag:
                print(f'epoch: {epoch}')
            
            if dataloader_train and (save_flag or train_flag):
                metrics_train = self.run_epoch(dataloader_train, phase='train', result_save_path=result_save_path, 
                                               gradient_accumulation_steps=gradient_accumulation_steps)
                if(epoch==0):
                    import subprocess
                    with open(result_save_path+'/val.txt', 'a') as f:
                        subprocess.run(["nvidia-smi"], stdout=f, text=True)
                
                print(f'train metrics of epoch {epoch}: {metrics_train}')
                with open(result_save_path+'/train.txt', 'a') as f:
                    f.write(f'train metrics of epoch {epoch}: {metrics_train}\n')

            if dataloader_val and (save_flag or train_flag):

                # 🌟🌟🌟 ここにお引越し：Valの直前にリセットし、ヘッダーを書く 🌟🌟🌟
                if hasattr(self.model, 'reset_topk_debug_flags'):
                    self.model.reset_topk_debug_flags()
                    self.model.result_save_path = result_save_path  # 👈【追加】モデルに保存先パスを教える
                    try:
                        # 👈【変更】保存先を result_save_path の中にする
                        log_file_path = f"{result_save_path}/topk_debug_log.txt"
                        with open(log_file_path, "a", encoding="utf-8") as f:
                            f.write(f"\n{'='*50}\n 🚀 Epoch {epoch} の 検証 Top-k 単語観測 (Validation)\n{'='*50}\n")
                    except Exception:
                        pass
                # 🌟🌟🌟 追加ここまで 🌟🌟🌟
                
                metrics_val, eval_data = self.run_epoch(dataloader_val, phase='eval', result_save_path=result_save_path)
                print(f'validation metrics of epoch {epoch}: {metrics_val}')

                with open(result_save_path+'/val.txt', 'a') as f:
                    f.write(f'validation metrics of epoch {epoch}: {metrics_val}\n')

                    # 判定を「val_lossが過去最小を下回ったか」に変更！
                    if metrics_val_best is None or metrics_val['val_loss'] < metrics_val_best.get('val_loss', float('inf')):
                        metrics_val_best = metrics_val
                        metrics_val_best['val_loss'] = metrics_val['val_loss']
                        epochs_no_improve = 0
                        print(f'new best validation metrics (by Validation Loss)')
                        f.write(f'new best validation metrics (by Validation Loss)\n')

                        if save_flag:
                            # ⬇️ 修正: 引数に epoch と metrics_val_best を追加
                            self.save(model_weight_path, weight_path, epoch=epoch, metrics_val_best=metrics_val_best)

                        (all_types_true, all_types_pred, all_times_true, all_time_preds, all_time_deltas_true, seq_scores_val) = eval_data
                        if self.temporal_emb_type in ['MLP_token_renew_vocab_topk_loss_ph','MLP_token_renew_vocab_loss_ph_scaled','MLP_token_renew_loss', 'MLP_token_renew_loss_p', 'MLP_token_renew_loss_h','MLP_token_renew_loss_ph','MLP_token_renew_topk_loss_p', 'MLP_token_renew_topk_loss_h', 'MLP_token_renew_topk_loss_ph','MLP_token_renew_vocab_loss_ph']:
                                    
                            quantitative_analysis_func(
                                all_types_true, all_types_pred, all_times_true, all_time_preds, all_time_deltas_true,
                                result_save_path, "val", epoch, self.model,seq_scores_val
                            )
                    else:
                        epochs_no_improve += 1
                        print(f'No improvement for {epochs_no_improve} epochs')
                        f.write(f'No improvement for {epochs_no_improve} epochs\n')

                if epochs_no_improve >= patience:
                    with open(result_save_path+'/val.txt', 'a') as f:
                        f.write(f"\nEarly stopping triggered at epoch {epoch}\n")
                    
                    # 🌟 ここで明示的にTest評価を行う（今のモデル状態で計算させる）
                    if dataloader_test:
                        print("Early stopping triggered. Running final test evaluation...")
                        metrics_test, _ = self.run_epoch(dataloader_test, phase='eval', result_save_path=result_save_path)
                        
                        import json
                        with open(result_save_path+'/test.txt', 'a') as f:
                            f.write(f'test metrics of epoch {epoch}: {metrics_test}\n')
                        print("Test metrics at stopping point saved.")
                        
                    break

            if dataloader_test and (save_flag or train_flag):
                # 🌟🌟🌟 ここにお引越し：Valの直前にリセットし、ヘッダーを書く 🌟🌟🌟
                if hasattr(self.model, 'reset_topk_debug_flags'):
                    self.model.reset_topk_debug_flags()
                    self.model.result_save_path = result_save_path  # 👈【追加】モデルに保存先パスを教える
                    try:
                        # 👈【変更】保存先を result_save_path の中にする
                        log_file_path = f"{result_save_path}/topk_debug_log.txt"
                        with open(log_file_path, "a", encoding="utf-8") as f:
                            f.write(f"\n{'='*50}\n 🚀 Epoch {epoch} の テスト Top-k 単語観測 (Validation)\n{'='*50}\n")
                    except Exception:
                        pass
                metrics_test, test_eval_data = self.run_epoch(dataloader_test, phase='eval', result_save_path=result_save_path)
                print(f'test metrics of epoch {epoch}: {metrics_test}')
                with open(result_save_path+'/test.txt', 'a') as f:
                    f.write(f'test metrics of epoch {epoch}: {metrics_test}\n')
